思路分析

经典的有点集，然后求解点集内部的最大曼哈顿距离。

考察曼哈顿距离的计算式为 $|x_1-x_2|+|y_1-y_2|$，有经典结论 $|x_1-x_2|+|y_1-y_2|=\max((+x_1+y_1)-(+x_2+y_2),(-x_1+y_1)-(-x_2+y_2),(+x_1-y_1)-(+x_2-y_2),(-x_1-y_1)-(-x_2-y_2))$，其正确性是显然的，就是在枚举 $x,y$ 这两维的加减分别的两种情况。

那么此时这个东西显然可以上线段树维护了。具体来说，我们维护 $x+y$ 的最大值、最小值，及其所属组编号，以及不属于这个组的点的 $x+y$ 的最大值、最小值，然后就可以把 $\max$ 的第一项的结果求出来了。

我们维护 $4$ 个类似这样的东西即可解决，时间复杂度 $O(n\log n)$，常数比较大，下面是一个非常可读的参考代码：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
int t,n,m,x[100005],y[100005],c[100005];
struct mx_pack{
	int v1,p1,v2;
	inline void reset(){ v1=v2=-1e10; }
	inline void set(int v,int p){
		v1=v; p1=p;
	}
	inline void ins(int v,int p){
		if(p==p1){ v1=max(v1,v); return; }
		if(v>v1) v2=v1,v1=v,p1=p;
		else v2=max(v2,v);
	}
	inline mx_pack operator+(const mx_pack&r)const{
		static mx_pack ret;
		ret=r;
		ret.ins(v1,p1);
		ret.ins(v2,0);
		return ret;
	}
};
struct mi_pack{
	int v1,p1,v2;
	inline void reset(){ v1=v2=1e10; }
	inline void set(int v,int p){
		v1=v; p1=p;
	}
	inline void ins(int v,int p){
		if(p==p1){ v1=min(v1,v); return; }
		if(v<v1) v2=v1,v1=v,p1=p;
		else v2=min(v2,v);
	}
	inline mi_pack operator+(const mi_pack&r)const{
		static mi_pack ret;
		ret=r;
		ret.ins(v1,p1);
		ret.ins(v2,0);
		return ret;
	}
};
struct pack{
	mx_pack xlu,xru,xld,xrd;
	mi_pack ilu,iru,ild,ird;
	inline void reset(){
		xlu.reset(); ilu.reset();
		xru.reset(); iru.reset();
		xld.reset(); ild.reset();
		xrd.reset(); ird.reset();
	}
	inline void set(int x,int y,int c){
		xlu.set(x+y,c); ilu.set(x+y,c);
		xru.set(x-y,c); iru.set(x-y,c);
		xld.set(-x+y,c); ild.set(-x+y,c);
		xrd.set(-x-y,c); ird.set(-x-y,c);
	}
	inline pack operator+(const pack&r)const{
		static pack ret;
		ret.xlu=xlu+r.xlu; ret.ilu=ilu+r.ilu;
		ret.xru=xru+r.xru; ret.iru=iru+r.iru;
		ret.xld=xld+r.xld; ret.ild=ild+r.ild;
		ret.xrd=xrd+r.xrd; ret.ird=ird+r.ird;
		return ret;
	}
};
inline int doit(const mx_pack&xp,const mi_pack&ip){
	if(xp.p1!=ip.p1) return xp.v1-ip.v1;
	else return max(xp.v2-ip.v1,xp.v1-ip.v2);
}
struct seg_tree{
	struct node{ int l,r; pack v; }re[100005<<2];
	inline void build(int l,int r,int p){
		re[p].l=l; re[p].r=r; re[p].v.reset();
		if(l==r) return;
		build((l+r>>1)+1,r,p<<1|1);
		build(l,(l+r>>1),p<<1);
	}
	inline void ins(int cp,int p){
		if(re[p].l==re[p].r) return re[p].v.set(x[cp],y[cp],c[cp]);
		ins(cp,p<<1|(cp>re[p<<1].r));
		re[p].v=re[p<<1].v+re[p<<1|1].v;
	}
	inline pack que(int l,int r,int p){
		if(re[p].l>=l&&re[p].r<=r) return re[p].v;
		if(r<=re[p<<1].r) return que(l,r,p<<1);
		if(l>re[p<<1].r) return que(l,r,p<<1|1);
		return que(l,r,p<<1)+que(l,r,p<<1|1);
	}
}sgt;
signed main(){
	ios::sync_with_stdio(0); cin>>t;
	for(int tt=1;tt<=t;++tt){
		cin>>n>>m; sgt.build(1,n,1);
		for(int i=1;i<=n;++i)
			cin>>x[i]>>y[i]>>c[i], sgt.ins(i,1);
		cout<<"Case #"<<tt<<":\n";
		for(int i=1,o,l,r,v;i<=m;++i)
			if(cin>>o>>l>>r,o==1)
				cin>>v,x[l]+=r,y[l]+=v,
				sgt.ins(l,1);
			else if(o==2) c[l]=r,sgt.ins(l,1);
			else{
				pack ret=sgt.que(l,r,1); int ans=0;
				ans=max(ans,doit(ret.xld,ret.ild));
				ans=max(ans,doit(ret.xlu,ret.ilu));
				ans=max(ans,doit(ret.xrd,ret.ird));
				ans=max(ans,doit(ret.xru,ret.iru));
				cout<<ans<<endl;
			}
	}
}
