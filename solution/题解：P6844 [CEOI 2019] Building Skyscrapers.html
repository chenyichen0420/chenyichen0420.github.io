思路分析

考试的时候理解错题意了，写了一大坨对完了的东西获得 0pts。

题意是按照一定的顺序建造楼房，要求后建的与之前建的八连通，建造的时候与外界连通而非最终与外界连通。问建造顺序反过来的字典序最大方案。

其实最后要求的东西已经强烈暗示我们需要逆序去做这个问题。于是显然问题变成了两个子问题：

  1. 如何判断一栋楼与外界连通？
  2. 如何判断一个点是不是割点？

第一个问题是相对简单的：我们可以先把所有楼房建起来，将外围的空地标记为外围连通，内部的空地分别使用不同的标记。显然我们只需要关注房子周围的八格内的空地即可，其他的不会造成影响。

当拆除一栋房子的时候，按照要求，他必须和外界连通，因此我们将和这栋房子挨着的内部空地重新标记为与外部连通即可。

显然，空地的数量是 $O(n)$ 的，且最多被重新标注一次，因此这一部分的时间复杂度可以暂且视为 $O(n)$。

第二个问题相对困难一些，不过由于房子是八连通的，因此我们不难列举出来可能的割点必然形如下面两种情况之一：

其中红点表示要删除的点，绿点和蓝点分别表示将要被割开的两个连通块，白点表示没有房子的位置。

更加细致的推导一下，我们发现：那两个白点必须是连通的。充分性是显然的，必要性是说如果不连通，则绿色和蓝色连通，不是真正的割点。

于是我们判断割点就变成了判断这个点是否满足上述两个形式中的某一个。具体来说就是绿色有点，蓝色有点，白色是相同空白连通块。需要注意：上面的形状可以旋转，因此实际上有 $6$ 个情况需要判断。

于是直接大力模拟就行了。下面是我写的被卡常的代码：

CPP 
#include<bits/stdc++.h>
using namespace std;
constexpr int inf=1e9+5,
dx[]={0,0,1,-1,1,1,-1,-1},
dy[]={1,-1,0,0,1,-1,1,-1};
int n,t,idc,f[200005]; pair<int,int>p[200005];
map<pair<int,int>,int>pid; int cnt=0,ed=0;
vector<int>ans; set<int,greater<int>>s;
bool inq[200005];
inline void dfs(const pair<int,int>&p){
    int cid=pid[p]; f[cid]=cid; pair<int,int>np; cnt++;
    for(int i=0;i!=8;++i){
        np=p;np.first+=dx[i],np.second+=dy[i];
        if(pid.count(np)&&!f[pid[np]]) dfs(np);
    }
}
struct validator{ vector<int>ep,ap,bp; }val[6];
inline bool split(const pair<int,int>&p,int vi){
    pair<int,int>np; validator&v=val[vi];
    np=p;np.first+=dx[v.ep[0]],np.second+=dy[v.ep[0]];
    int ed1=pid[np]; if(ed1>0) return 0;
    np=p;np.first+=dx[v.ep[1]],np.second+=dy[v.ep[1]];
    int ed2=pid[np]; if(ed2>0) return 0;
    if(ed1!=ed2) return 0;
    bool hsa=0,hsb=0;
    for(int i:v.ap){
        np=p;np.first+=dx[i],np.second+=dy[i];
        if(pid[np]>0){hsa=1;break;}
    }
    for(int i:v.bp){
        np=p;np.first+=dx[i],np.second+=dy[i];
        if(pid[np]>0){hsb=1;break;}
    }
    return hsa&&hsb;
}
inline bool valid(const pair<int,int>&p){
    bool hso=0; pair<int,int>np; 
    for(int i=0;i!=4;++i){
        np=p;np.first+=dx[i],np.second+=dy[i];
        if(pid[np]==-1){hso=1;break;}
    }
    if(!hso) return 0;
    for(int i=0;i!=6;++i)
        if(split(p,i)) return 0;
    return 1;
}
inline void dfe(const pair<int,int>&p,bool ck=0){
    pid[p]=ed; pair<int,int>np;
    for(int i=0;i!=4;++i){
        np=p;np.first+=dx[i],np.second+=dy[i];
        if(pid.count(np)&&pid[np]!=ed&&pid[np]<=0) dfe(np,ck);
    }
    if(ck){
        for(int j=0;j!=8;++j){
            int id=pid[{p.first+dx[j],p.second+dy[j]}];
            if(id>0){
                if(!inq[id]&&valid(::p[id])) s.emplace(id),inq[id]=1;
                else if(inq[id]&&!valid(::p[id])) s.erase(id),inq[id]=0;
            }
        }
    }
}
signed main(){
    ios::sync_with_stdio(0); cin>>n>>t;
    for(int i=1;i<=n;++i)
        cin>>p[i].first>>p[i].second,pid[p[i]]=i;
    if(dfs(p[1]),cnt!=n) return puts("NO"),0;
    for(int i=1;i<=n;++i)
        for(int j=0;j!=8;++j)
            if(!pid.count({p[i].first+dx[j],p[i].second+dy[j]}))
                pid[{p[i].first+dx[j],p[i].second+dy[j]}]=0;
    for(const auto&v:pid) if(v.second==0) ed--,dfe(v.first);
    val[0].ep={0,1}; val[0].ap={5,2,4}; val[0].bp={7,3,6};
    val[1].ep={2,3}; val[1].ap={5,1,7}; val[1].bp={4,0,6};
    val[2].ep={1,2}; val[2].ap={5}; val[2].bp={4,0,6,3,7};
    val[3].ep={2,0}; val[3].ap={4}; val[3].bp={6,3,7,1,5};
    val[4].ep={0,3}; val[4].ap={6}; val[4].bp={7,1,5,2,4};
    val[5].ep={3,1}; val[5].ap={7}; val[5].bp={5,2,4,0,6};
    for(int i=1;i<=n;++i) if(valid(p[i])) s.emplace(i),inq[i]=1;
    for(ed=-1;s.size();){
        int v=*s.begin(); s.erase(s.begin()); inq[v]=0;
        ans.emplace_back(v); dfe(p[v],1);
    }
    reverse(ans.begin(),ans.end()); cout<<"YES\n";
    for(int v:ans) cout<<v<<"\n";
}
