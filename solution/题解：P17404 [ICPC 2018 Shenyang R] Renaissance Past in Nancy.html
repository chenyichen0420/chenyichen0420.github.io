思路分析

对生成函数完全不熟练啊，需要多练一些生成函数题了。

对于第 $i$ 个物品，显然其生成函数为 $1+x^{b_i}+x^{2b_i}+\dots+x^{a_ib_i}$。而根据经典的变换，上式实际上就是 $\frac{1-x^{b_i(a_i+1)}}{1-x^{b_i}}$。

于是我们询问的 $[l,r]$ 区间内的所有物品的生成函数实际上就是 $\prod_{i=l}^r\frac{1-x^{b_i(a_i+1)}}{1-x^{b_i}}$。

我们记 $f_i(x)=\frac{1-x^{b_i(a_i+1)}}{1-x^{b_i}}$，所求即为 $\prod_{i=l}^rf_i$。

显然我们可以考虑使用前缀积、前缀逆元的形式来优化这个东西。具体来说，我们令 $F_i(x)=\prod_{j=1}^if_j(x)$，则所求即为 $\sum_{i=0}^c[x^i](F_r*F_{l-1}^{-1})=[x^c](\frac{1}{1-x}*F_r*F_{l-1}^{-1})$。

于是我们考虑预处理两个东西：$\frac{1}{1-x}*F_i$ 和 $F_i^{-1}$。

我们以 $F_i$ 为例，考虑怎么做：

$f_i=\frac{1-x^{b_i(a_i+1)}}{1-x^{b_i}}=(1-x^{b_i(a_i+1)})\times\frac{1}{1-x^{b_i}}=(1-x^{b_i(a_i+1)})(1+x^{b_i}+x^{2b_i}+\dots)$。

显然第一项等价于做一个大小为 $b_i(a_i+1)$ 的逆零一背包，也就是从背包中删去大小为 $b_i(a_i+1)$ 的东西。第二项等价于做一个大小为 $b_i$ 的完全背包。

于是我们可以 $O(V)$ 的从 $F_{i-1}$ 推到 $F_i$。

原先所要求的两个东西的具体推导过程与计算方式都是类似的，在此不再进行展开。

于是时间复杂度为 $O(TnV)$，足以通过此题。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
constexpr int mod=1e9+7;
int t,n,m,a[10005],b[10005],f[10005][1005],g[10005][1005];
inline void mad(int&l,const int r){(l+=r)>=mod&&(l-=mod);}
inline void mdc(int&l,const int r){(l-=r)<0&&(l+=mod);}
signed main(){
	ios::sync_with_stdio(0); cin>>t;
	for(int tt=1;tt<=t;++tt){
		for(int i=0;i<=n;++i)
			memset(f[i],0,sizeof f[i]),
			memset(g[i],0,sizeof g[i]);
		cin>>n>>m; f[0][0]=g[0][0]=1;
		for(int i=1;i<=n;++i) cin>>a[i]>>b[i];
		for(int i=1;i<=n;++i){
			memcpy(f[i],f[i-1],sizeof f[0]);
			for(int j=b[i];j<=1000;++j)
				mad(f[i][j],f[i][j-b[i]]);
			for(int j=1000;j>=b[i]*(a[i]+1);--j)
				mdc(f[i][j],f[i][j-b[i]*(a[i]+1)]);
		}
		for(int i=0;i<=n;++i)
			for(int j=1;j<=1000;++j)
				mad(f[i][j],f[i][j-1]);
		for(int i=1;i<=n;++i){
			memcpy(g[i],g[i-1],sizeof g[0]);
			for(int j=b[i]*(a[i]+1);j<=1000;++j)
				mad(g[i][j],g[i][j-b[i]*(a[i]+1)]);
			for(int j=1000;j>=b[i];--j)
				mdc(g[i][j],g[i][j-b[i]]);
		}
		cout<<"Case #"<<tt<<":\n";
		for(int i=1,l,r,c,la=0;i<=m;++i){
			cin>>l>>r>>c;
			l=(l+la)%n+1;
			r=(r+la)%n+1;
			if(l>r) swap(l,r);
			la=0;
			for(int j=0;j<=c;++j)
				la=(la+f[r][j]*g[l-1][c-j])%mod;
			cout<<la<<endl;
		}
	}
}
