思路分析

区间 DP 上树的模板题？

考虑如果是一个序列，我们怎么做：显然我们设 $dp_{i,j}$ 表示区间 $[i,j]$ 内部的答案。然后转移为 $dp_{i,j}=\max(dp_{i,j-1},dp_{i+1,j},dp_{i+1,j-1}+[c_i=c_j]\times2)$。

考虑这个过程上树应该怎么办。我们发现解决方法似乎没有什么本质区别。我们仍然可以考虑 $dp_{i,j}$ 按照类似于上面的向内收缩的方式进行转移。方程会变成下面这个样子：

$dp_{i,j}=\max(dp_{fa_{j,i},j},dp_{i,fa_{i,j}},dp_{fa_{j,i},fa_{i,j}}+[c_i=c_j]\times2)$

其中的 $fa_{i,j}$ 表示以 $i$ 为根的情况下 $j$ 的父节点。于是此题可解，时间复杂度 $O(n^2)$。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
vector<int>son[2005];
int t,n,dp[2005][2005],fa[2005][2005],ans; char c[2005];
vector<pair<int,int>>dr[2005];
inline void dfs(int p,int f,int d,int rt){
	fa[rt][p]=f;
	dr[d].emplace_back(p,rt);
	for(int sp:son[p])
		if(sp!=f) dfs(sp,p,d+1,rt);
}
signed main() {
	ios::sync_with_stdio(0);
	for(cin>>t;t;t--){
		cin>>n>>c; memmove(c+1,c,n);
		for(int i=1;i<=n;++i)
			son[i].clear(),dr[i].clear();
		for(int i=1;i<=n;++i)
			for(int j=1;j<=n;++j)
				dp[i][j]=0;
		for(int i=1;i<=n;++i) dp[i][i]=1;
		for(int i=1,l,r;i!=n;++i)
			cin>>l>>r,
			son[l].emplace_back(r),
			son[r].emplace_back(l),
			dp[l][r]=dp[r][l]=1+(c[l]==c[r]);
		for(int i=1;i<=n;++i) dfs(i,0,0,i);
		for(int i=2;i<=n;++i)
			for(const auto&v:dr[i])
				dp[v.first][v.second]=max({
					dp[v.first][fa[v.first][v.second]],
					dp[fa[v.second][v.first]][v.second],
					dp[fa[v.second][v.first]][fa[v.first][v.second]]
					+(c[v.first]==c[v.second])*2
				});
		for(int i=1;i<=n;++i)
			for(int j=1;j<=n;++j)
				ans=max(ans,dp[i][j]);
		cout<<ans<<endl; ans=0;
	}
}
