思路分析

看了一眼数据范围，非常像是 $3\times\log_2n$，于是我们思考一下怎么进行分治。

我们想要找的数字是 $1$，$1$ 有什么性质是别的数不具有的？对于任何的 $k$（显然我们可以忽略毫无用处的 $k=1$），$\lfloor\frac{1}{k}\rfloor=0$。

这样的性质在 $k=2$ 的时候是最为明显的，因为此时对于任何 $v\neq 1,\lfloor\frac{v}{k}\rfloor\neq0$。

于是我们考虑，如果使用 $k=2$ 的询问，我们可以得到什么样的信息。

一个自然的观察是那些结果一样的数字形如 $2v,2v+1$，也就可以理解为这样的数字两两配对。

如果我们选择一个分割点 $p$，我们询问 $[1,p],k=2$ 和 $[p+1,n],k=2$，我们就可以知道两边有多少个不同的数字，进而可以知道两边有多少个数字是没有配对的。

对于大多数情况，如果左边有一个没有配对的数字，那么原本和他配对的数字就会出现在右边。也就是说，两边的没有配对的数字的数量应该是近似相等的。

什么数字导致了两边不相等呢？$1$ 是一个，他是一定没有配对的。还有，如果 $n$ 是偶数的话，他也一定是没有配对的。

于是我们可以考虑利用这一点进行二分。较为简单的，我们可以先考虑 $n$ 是奇数的情况。如果当前答案区间是 $[l,r]$ 的话，那么询问 $[1,mid],k=2$ 和 $[mid+1,n],k=2$ 并计算未配对的数的个数。

如果左边的更多（应该恰好多一），那么 $1$ 在左边，$r=mid$，反之 $1$ 在右边，$l=mid+1$。

现在考虑偶数的情况。上面的过程显然需要消耗 $2\times\log_2n$ 的操作次数。于是我们考虑 $n$ 又有什么特殊性质？

显然，只有 $n$ 满足 $\lfloor\frac{n}{n}\rfloor=1$，别的 $v\neq n$ 都有 $\lfloor\frac{v}{n}\rfloor=0$。

于是显然我们可以使用一个 $\log$ 把 $n$ 的位置二分出来。具体来说，当答案区间是 $[l,r]$ 的时候，我们可以询问 $[1,mid],k=n$，如果是 $2$ 则说明 $n$ 在 $[l,mid]$，否则说明答案在 $[mid+1,r]$。

于是显然这就是 $3\times\log_2n$ 的做法。

还能更优秀一点吗？

我们考虑在 $n$ 是奇数的情况下的那个做法什么时候会爆炸：显然就是 $1$ 和 $n$ 相对于 $mid$ 一左一右的时候。

那么，我们考虑是不是可以等用到的时候再确认一下到底是在哪一边呢？

确实可以，确认的方式和前面是一样的，不再赘述，注意我们只需要确认一次，因为区间是缩小的。确认之后记录一下 $n$ 是在区间的左边还是右边就行了。

于是总的操作次数为 $2\times\log_2n+1$，可以通过 $1$ 和 $2$。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
// #define DBG
#ifdef DBG
	string input="113 1 111 63 101 51 68 81 11 105 28 36 7 67 4 14 65 3 22 71 39 99 104 78 40 98 45 66 10 92 64 46 69 30 112 42 27 49 19 32 34 43 93 85 55 74 41 72 8 79 109 100 88 23 107 62 53 5 89 87 9 18 17 31 50 33 95 37 57 48 76 24 26 82 103 35 77 47 60 61 75 86 110 94 70 16 13 58 84 80 59 52 56 114 44 90 102 83 91 25 12 29 108 20 73 2 21 106 15 38 6 97 54 96";
	int a[115];
#endif
int n,l,r,mid,vl,vr,sd;
inline int query(int l,int r,int k){
	cout<<"? "<<l<<" "<<r<<" "<<k<<endl;
#ifdef DBG
	set<int>s;
	for(int i=l;i<=r;++i)
		s.emplace(a[i]/k);
	cout<<s.size()<<endl;
	return s.size();
#endif
	int ret; cin>>ret; return ret;
}
signed main(){
	ios::sync_with_stdio(0); 
#ifdef DBG
	vector<int>vv; int vt;
	istringstream iss(input);
	while(iss>>vt) vv.emplace_back(vt);
	l=1, n=r=vv.size();
	for(int i=1;i<=n;++i) a[i]=vv[i-1];
#else
	cin>>n; l=1,r=n;
#endif
	while(l!=r){
		mid=l+r>>1;
		vl=query(1,mid,2);
		vr=query(mid+1,n,2);
		vl=vl*2-mid;
		vr=vr*2-(n-mid);
		if(sd)
			if(sd==1) vl--;
			else vr--;
		if(vl>vr) r=mid;
		else if(vl<vr) l=mid+1;
		else{
			if(mid!=1){
				if(query(1,mid,n)==1) r=mid,sd=2;
				else l=mid+1,sd=1;
			}
			else{
				if(query(mid+1,n,n)==1) l=mid+1,sd=1;
				else r=mid,sd=2;
			}
		}
		cerr<<l<<" "<<r<<endl;
	}
	cout<<"! "<<l<<endl;
#ifdef DBG
	if(a[l]!=1) cout<<"Incorrect!\n";
	else cout<<"Correct!\n";
#endif
	return 0;
}
