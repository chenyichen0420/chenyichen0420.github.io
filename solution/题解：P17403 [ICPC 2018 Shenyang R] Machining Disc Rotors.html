思路分析

我们考虑将所有的圆都切了之后，边界上会变得缺很多块。但是好在题目保证了切的部分是不相交的。我们定义这些被切下来的部分与原边界的交点为界点。

显然，如果切完之后仍然存在直径，那么直接选直径就是最优的。

至于怎么判断直径：直接枚举界点，对侧的点显然就是横纵坐标都选择相反数，随后我们枚举所有的圆，检查这个点在不在园内即可。这是 $O(n^2)$ 的。

随后，根据一点人类智慧，答案只会是界点之间的连线。证明是简单的：

如果答案在图形内部一定不优秀，所以答案一定在边界上。

如果答案在一个被切出来的圆弧上，那么这一定不同时比放到这个圆弧的两个界点上面优秀。因为另一个点在原来的大圆内部。我们考虑以这个点为圆心进行画圆。那么当这个圆覆盖了两个界点的时候，由于圆心在原来的大圆内，因此我画出来的这个圆一定已经完全包含了切出来的整个圆弧。

如果答案一端在某一段原先的圆弧上，由于直径被切掉了，因此我总是能够向着直径那个方向移动这一端，将答案变大。

因此答案一定出现在界点的连线中。显然这仍然是 $O(n^2)$ 的。

于是此题可解，时间复杂度 $O(n^2)$。剩下的求两个圆的交点只是一些简单的 dirty work，在此不赘述，参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define double long double
int t,n,m;
struct dot{double x,y;};
struct circle{dot p;double r;}mc;
inline double dis(const dot&l,const dot&r){
	return sqrtl((l.x-r.x)*(l.x-r.x)+(l.y-r.y)*(l.y-r.y));
}
inline bool eq(double l,double r){ return abs(l-r)<=1e-9; }
inline bool incirc(const dot&p,const circle&c){ return dis(p,c.p)<=c.r+1e-9; }
vector<dot>cirint(const circle&l,const circle&r){
	vector<dot>ret; double d=dis(l.p,r.p);
	if(d>l.r+r.r+1e-9||d<abs(l.r-r.r)-1e-9) return ret;
	if(eq(d,l.r+r.r)||eq(d,abs(l.r-r.r))) return ret;
	double a=(l.r*l.r-r.r*r.r+d*d)/(2*d);
	double h=sqrtl(max(0.0l,l.r*l.r-a*a));
	dot p1,p2,p3;
    p2.x=l.p.x+a*(r.p.x-l.p.x)/d;
    p2.y=l.p.y+a*(r.p.y-l.p.y)/d;
    p1.x=p2.x+h*(r.p.y-l.p.y)/d;
    p1.y=p2.y-h*(r.p.x-l.p.x)/d;
    p3.x=p2.x-h*(r.p.y-l.p.y)/d;
    p3.y=p2.y+h*(r.p.x-l.p.x)/d;
	ret.emplace_back(p1);
	ret.emplace_back(p3);
	return ret;
}
const double pi=acosl(-1);
vector<dot>cup; vector<circle>cuc;
signed main(){
	ios::sync_with_stdio(0); cin>>t;
	for(int tt=1;tt<=t;++tt){
		cin>>n>>m;
		cup.clear(); cuc.clear();
		mc.p.x=mc.p.y=0; mc.r=m;
		for(int i=1,x,y,r;i<=n;++i){
			cin>>x>>y>>r; circle c;
			c.p.x=x; c.p.y=y; c.r=r;
			vector<dot>inp=cirint(mc,c);
			if(!inp.size()) continue;
			cuc.emplace_back(c);
			for(const auto&v:inp) cup.emplace_back(v);
		}
		if(!cup.size()) printf("Case #%d: %.10Lf\n",tt,2.0*m);
		else{
			double ans=0;
			for(const auto&v:cup){
				for(const auto&rv:cup)
					ans=max(ans,dis(v,rv));
				dot tv=v; tv.x=-tv.x; tv.y=-tv.y; bool cn=0;
				for(const auto&cv:cuc)
					if(incirc(tv,cv)){ cn=1; break; }
				if(!cn){ ans=m*2; break; }
			}
			printf("Case #%d: %.10Lf\n",tt,ans);
		}
	}
}
