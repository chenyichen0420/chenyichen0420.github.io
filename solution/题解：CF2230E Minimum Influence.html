思路分析

一个非常显然的想法是把所有情况直接分类讨论了，拆成 $9$ 种情况，硬上数据结构。

是对的，但是应该过不去，常数都太大了，我们思考有没有更优秀一点的做法。

我们注意一个性质：$I_p\le p_i$，也就是说如果我把某一个贡献的式子中的一部分 $I_p$ 替换成 $p_i$ 进行统计的话是不会导致答案算小的。

考虑我们的九种情况当中，角落上的四种都只要求判断范围内有没有点，显然可以使用扫描线直接线性的解决掉（值域范围有限）。

最中间的一种情况是求解 $p,c$ 分别在一定范围内的点的 $p_i+c_i$ 的最小值，使用前面的性质进行放大，我们可以使用全局的 $p_i+c_i$ 的最小值代替范围内的最小值，也是线性的。

边上的四种要求求出一个一维有上下界，一维有上界或者下界的区域内的某一个维上的最小值。

注意到在有上下界的那一维中，我们对于上界以上的部分使用放大，即可将上界抹除掉，此时问题扫描先后变成了求解后继，使用 set 做一下就行。

总时间复杂度是 $O(V+n\log n)$ 的，其中 $V$ 是 $p,c$ 的值域，$n,m$ 同阶。参考代码如下：

CPP 
#include <bits/stdc++.h>
using namespace std;
#pragma GCC optimize(2)
struct IO {
#define mxsz (1 << 21)
	char buf[mxsz], * p1, * p2;
	char pbuf[mxsz], * pp; bool acc[128];
	IO() : p1(buf), p2(buf), pp(pbuf) {}
	~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
	inline char gc() {
		if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, mxsz, stdin);
		return p1 == p2 ? ' ' : *p1++;
	}
	inline void setacc(const char* c) { while (*c) acc[*c++] = 1; }
	inline char getc() { char c; while (!acc[c = gc()]); return c; }
#ifndef sipt
	inline int read() {
		int r = 0; char c = gc(); while (c < '0' || c>'9') c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return r;
	}
#else
	inline int read() {
		int r = 0; char c = gc(); bool rev = 0;
		while (c < '0' || c>'9') rev |= (c == '-'), c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return rev ? ~r + 1 : r;
	}
#endif
	inline void push(const char& c) {
		if (pp - pbuf == mxsz) fwrite(pbuf, 1, mxsz, stdout), pp = pbuf;
		*pp++ = c;
	}
	inline void write(int x) {
		static char sta[22]; int top = 0;
		do sta[top++] = x % 10, x /= 10; while (x);
		while (top) push(sta[--top] ^ 48);
	}
	inline void write(int x, char opc) {
#ifdef sopt
		if (x < 0) push('-'), x = ~x + 1;
#endif
		write(x), push(opc);
	}
	inline void puts(const char* c) { while (*c) push(*c++); }
} io;
//私は猫です!
int n, m, tp[2000005], tc[2000005], d[2000005];
int ans[2000005], mia, vmi, vma; pair<int, int>p[2000005];
set<int>s;
struct line {
	int l, r, tar, ty;
}; vector<line>lc[2000005];
signed main() {
	ios::sync_with_stdio(0); n = io.read();
	for (int i = 1; i <= n; ++i) p[i].first = io.read();
	for (int i = 1; i <= n; ++i) p[i].second = io.read();
	mia = 1e9; m = io.read();
	for (int i = 1; i <= m; ++i) tp[i] = io.read();
	for (int i = 1; i <= m; ++i) tc[i] = io.read();
	for (int i = 1; i <= m; ++i) d[i] = io.read();
	/*
	每一个人在 p,c 平面上划分出九个区间
	四角区间只关心有没有元素
	四边区间只关心某一维的最小值
	中心区间只关心两维之和的最小值
	中间的恰为 ai+bi，别的地方都小于此上界
	此题可解，时间复杂度 nlogn
	*/
	for (int i = 1; i <= n; ++i)
		mia = min(mia, p[i].first + p[i].second);
	for (int i = 1; i <= m; ++i) ans[i] = mia;

	vmi = 1e9, vma = -1;
	for (int i = 1; i <= m; ++i)
		lc[tp[i]].push_back({ tc[i],tc[i] + d[i], i,1 }),
		lc[tp[i]].push_back({ tc[i],0,i,2 }),
		lc[tp[i]].push_back({ tc[i] + d[i],0,i,3 });
	for (int i = 1; i <= n; ++i)
		lc[p[i].first].push_back({ p[i].second,0,i,0 });
	s.clear(); s.emplace(1e9);
	for (int i = 0; i <= 2e6 + 2; ++i) {
		for (const auto& cs : lc[i]) {
			if (!cs.ty) s.emplace(cs.l), vmi = min(vmi, cs.l), vma = max(vma, cs.l);
			else if (cs.ty == 1) ans[cs.tar] = min(ans[cs.tar], *s.lower_bound(cs.l));
			else if (cs.ty == 2) {
				if (vmi < cs.l) ans[cs.tar] = 0;
			}
			else if (cs.ty == 3) {
				if (vma > cs.r) ans[cs.tar] = min(ans[cs.tar], tc[cs.tar] + d[cs.tar]);
			}
		}
	}
	for (int i = 1; i <= m; ++i)
		lc[tp[i]].pop_back(),
		lc[tp[i]].pop_back(),
		lc[tp[i]].pop_back();
	for (int i = 1; i <= n; ++i)
		lc[p[i].first].pop_back();

	vmi = 1e9, vma = -1;
	for (int i = 1; i <= m; ++i)
		lc[tp[i] + d[i]].push_back({ tc[i],tc[i] + d[i], i,1 }),
		lc[tp[i] + d[i]].push_back({ tc[i],0,i,2 }),
		lc[tp[i] + d[i]].push_back({ tc[i] + d[i],0,i,3 });
	for (int i = 1; i <= n; ++i)
		lc[p[i].first].push_back({ p[i].second,0,p[i].second,0 });
	s.clear(); s.emplace(1e9);
	for (int i = 2e6 + 2; i >= 0; --i) {
		for (const auto& cs : lc[i]) {
			if (!cs.ty) s.emplace(cs.l), vmi = min(vmi, cs.l), vma = max(vma, cs.l);
			else if (cs.ty == 1) ans[cs.tar] = min(ans[cs.tar], *s.lower_bound(cs.l) + tp[cs.tar] + d[cs.tar]);
			else if (cs.ty == 2) {
				if (vmi < cs.l) ans[cs.tar] = min(ans[cs.tar], tp[cs.tar] + d[cs.tar]);
			}
			else if (cs.ty == 3) {
				if (vma > cs.r) ans[cs.tar] = min(ans[cs.tar], tc[cs.tar] + d[cs.tar] + tp[cs.tar] + d[cs.tar]);
			}
		}
	}
	for (int i = 1; i <= m; ++i)
		lc[tp[i] + d[i]].pop_back(),
		lc[tp[i] + d[i]].pop_back(),
		lc[tp[i] + d[i]].pop_back();
	for (int i = 1; i <= n; ++i)
		lc[p[i].first].pop_back();
	for (int i = 1; i <= m; ++i)
		lc[tc[i]].push_back({ tp[i],tp[i] + d[i], i,1 });
	for (int i = 1; i <= n; ++i)
		lc[p[i].second].push_back({ p[i].first,0,p[i].first,0 });
	s.clear(); s.emplace(1e9);
	for (int i = 0; i <= 2e6 + 2; ++i) {
		for (const auto& cs : lc[i]) {
			if (!cs.ty) s.emplace(cs.l);
			else if (cs.ty == 1) ans[cs.tar] = min(ans[cs.tar], *s.lower_bound(cs.l));
		}
	}
	for (int i = 1; i <= m; ++i)
		lc[tc[i]].pop_back();
	for (int i = 1; i <= n; ++i)
		lc[p[i].second].pop_back();
	for (int i = 1; i <= m; ++i)
		lc[tc[i] + d[i]].push_back({ tp[i],tp[i] + d[i], i,1 });
	for (int i = 1; i <= n; ++i)
		lc[p[i].second].push_back({ p[i].first,0,p[i].first,0 });
	s.clear(); s.emplace(1e9);
	for (int i = 2e6 + 2; i >= 0; --i) {
		for (const auto& cs : lc[i]) {
			if (!cs.ty) s.emplace(cs.l);
			else if (cs.ty == 1) ans[cs.tar] = min(ans[cs.tar], *s.lower_bound(cs.l) + tp[cs.tar] + d[cs.tar]);
		}
	}

	for (int i = 1; i <= m; ++i) io.write(ans[i], '\n');
}
