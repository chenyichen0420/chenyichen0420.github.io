思路分析

其实看到几个数字的和的时候，就应该往卷积那方面猜了。

等价于 $x-a_i=a_j+a_k$，于是我们考虑枚举 $a_i$，同时利用卷积计算出 $a_j+a_k=v$ 的方案数。

后面那一步其实是简单的：我们考虑每出现一个 $a$，就将 $x^a$ 的系数增加 $1$，得到的多项式 $f(x)$，我们计算其平方 $[x^v]f^2(x)$ 即为 $a_j+a_k=v$ 的方案数。但是这并不是我们需要的数值。我们需要的是满足 $j

不过这是简单的。我们令 $g(x)=f^2(x)$，则显然 $[x^{2a_i}]g(x)$ 应该减少 $1$，减完之后我们再给每一项的系数除以 $2$ 即可得到我们想要的东西。

不过这还没有完，我们还需要在枚举 $i$ 的时候是否存在 $j,k\neq i$ 满足 $a_j+a_k=x-a_i$，不过这似乎是简单的。

此时再重复的话，显然至少出现了两个 $a_i$，于是将次数减少 $x-2\times a_i$ 在 $a$ 中的出现次数。

但是特别注意：如果 $3\times a_i=x$，则显然我们多去了一次，于是再 $+1$ 即可。

此时我们就得到了 $a_j+a_k=x-a_i$ 的方案数。如果这个方案数不为 $0$，我们就可以去找 $j,k$ 了。

实现也很简单：直接枚举 $j$，如果 $a$ 中出现过 $x-a_i-a_j$，那么再枚举 $k$。这样做其实仍然是 $O(n)$ 的，因为一个数字重复出现，导致误认为存在 $k$ 的情况的出现次数显然最多只有 $1$。

此题可解，时间复杂度 $O(V\log V+n)$，参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
// #define int long long
constexpr int mod=998244353;
using cpx=complex<double>;
int n,m,a[1000005],lim,cp,ac[1000005]; vector<cpx>v; vector<int>sp;
using cpx=complex<double>;
const double pi = acos(-1);
inline void fft(int lim,vector<cpx>&v,const vector<int>&sp) {
	for (int i = 0; i < lim; ++i)
		if (i < sp[i]) swap(v[i], v[sp[i]]);
	for (int ml = 1; ml < lim; ml <<= 1) {
		cpx tv(cos(pi / ml), sin(pi / ml));
		for (int p = 0; p < lim; p += ml << 1) {
			cpx vl(1, 0), tmp;
			for (int i = 0; i < ml; ++i, vl *= tv)
				tmp = v[p + i + ml] * vl, 
				v[p + i + ml] = v[p + i] - tmp,
				v[p + i] += tmp;
		}
	}
}
inline void out(int i,int j,int k){
	cout<<min({i,j,k})<<" "<<i+j+k-min({i,j,k})-max({i,j,k})<<" "<<max({i,j,k})<<endl;
	exit(0);
}
signed main(){
	lim=1; while(lim<1e6) lim<<=1,cp++;
	sp.resize(lim<<1); v.resize(lim<<1);
	ios::sync_with_stdio(0); cin>>n>>m;
	for(int i=1;i!=lim<<1;++i)
		sp[i]=(sp[i>>1]>>1)|((i&1)<<cp);
	for(int i=1;i<=n;++i) cin>>a[i],v[a[i]]+=1,ac[a[i]]++;
	fft(lim*2,v,sp); for(auto&a:v) a*=a; fft(lim*2,v,sp);
	reverse(v.begin()+1,v.begin()+lim*2);
	for(int i=0;i<=2e6;++i) sp[i]=v[i].real()/lim/2+1e-3;
	for(int i=1;i<=n;++i) sp[a[i]*2]--;
	for(int i=0;i<=2e6;++i) sp[i]>>=1;
	for(int i=1;i<=n;++i){
		if(a[i]>=m) continue;
		int cnt=sp[m-a[i]];
		if(a[i]*2<m) cnt-=ac[m-2*a[i]];
		if(a[i]*3==m) cnt++;
		if(cnt)
			for(int j=1;j<=n;++j)
				if(i!=j&&m-a[i]-a[j]>0&&ac[m-a[i]-a[j]])
					for(int k=1;k<=n;++k) if(a[i]+a[j]+a[k]==m)
						if(k!=i&&k!=j) out(i,j,k);
	}
	cout<<"-1\n";
}
