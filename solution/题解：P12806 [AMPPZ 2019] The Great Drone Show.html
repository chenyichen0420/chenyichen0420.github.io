思路分析

我非得每天听大狗叫吗？

突破口在于两点：

  1. 所有的电线的投影是不交的，也就是说整张图是一个简单平面图。
  2. 总边数最大只有 $3n$。

于是，任意时刻，在整张图当中必定存在点，使得其度数不超过 $5$。我们将这些点连的边定为外向边（也就是指出去，后面会解释），随后删除这个点和这些边。重复操作直到图被删空。

于是我们得到了一个有向图。我们想要做的事情是说，每一次发生更改的时候，我们只主动更新一个点的出边所指向的点的信息，而不主动接受一个点的入边带来的信息。

如果我们可以做到这一点的话，显然时间复杂度就是每一次 $O(1)$ 的了（实际上最多就是 $5$ 个点）。

事实上，我们的确可以做到，思考流程如下：

首先，对于出边，我们直接在更新的时候进行暴力检验就行了，这显然不是瓶颈。重点在于我应该怎么保证每一条边只被删除一次，尤其是入边被动删除。

注意到一个非常强的性质：无人机的横纵坐标不变，也就是说每一个电缆的限制都可以描述为两个无人机的高差不超过指定值。这个值可以 $O(1)$ 预处理出来。

于是对于每一个节点 $u$，我们要求指向他的节点 $v$ 一直满足 $h_v-\Delta h\le h_u\le h_v+\Delta h$。

于是我们直接在每一个节点维护两个 set，一个是指向他的节点的 $(h_v-\Delta h,edgeid_{u,v})$，另一个是 $(h_v+\Delta h,edgeid_{u,v})$。这样我们就可以快速遍历所有不合法的入边并断开了。

此外，这个信息的更新也是简单的，我们直接遍历出边的时候删除就的元素并添加新的元素就行了。

做完这一步，难点就没了，后面就是将整个过程倒过来，按照删边的时间顺序倒序合并所有的连通块。

直接建立一个类似什么 kruskal 重构树的结构出来，然后把答案拍上去，每一次查询的时候求一个 LCA 就行了。

每一步的时间复杂度都是单个 $\log$，不过我实现的常数比较大。参考代码如下：

#include<bits/stdc++.h>

struct IO {
#define mxsz (1 << 21)
	char buf[mxsz], * p1, * p2;
	char pbuf[mxsz], * pp; bool acc[128];
	IO() : p1(buf), p2(buf), pp(pbuf) {}
	~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
	inline char gc() {
		if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, mxsz, stdin);
		return p1 == p2 ? ' ' : *p1++;
	}
	inline void setacc(const char* c) { while (*c) acc[*c++] = 1; }
	inline char getc() { char c; while (!acc[c = gc()]); return c; }
#ifndef sipt
	inline int read() {
		int r = 0; char c = gc(); while (c < '0' || c>'9') c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return r;
	}
#else
	inline int read() {
		int r = 0; char c = gc(); bool rev = 0;
		while (c < '0' || c>'9') rev |= (c == '-'), c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return rev ? ~r + 1 : r;
	}
#endif
	inline void push(const char& c) {
		if (pp - pbuf == mxsz) fwrite(pbuf, 1, mxsz, stdout), pp = pbuf;
		*pp++ = c;
	}
	inline void write(int x) {
		static char sta[22]; int top = 0;
		do sta[top++] = x % 10, x /= 10; while (x);
		while (top) push(sta[--top] ^ 48);
	}
	inline void write(int x, char opc) {
#ifdef sopt
		if (x < 0) push('-'), x = ~x + 1;
#endif
		write(x), push(opc);
	}
	inline void puts(const char* c) { while (*c) push(*c++); }
} io;
#else
struct IO{

		int v; cin>>v; return v;

constexpr int doc=5e5+5,edc=1.5e6+5,moc=1e6+5;
int t,n,m,k,x[doc],y[doc],h[doc],l[edc],r[edc];
struct cable{int p,id,v;}; vector<cable>cab[doc],oca[doc];
pair<int,int>tcut[edc]; int csd[edc],du[doc]; queue<int>q;
inline int calr(int l,int r,int v){
	v=v*v-(x[l]-x[r])*(x[l]-x[r])-(y[l]-y[r])*(y[l]-y[r]);
	int ret=sqrt(v); while(ret*ret<=v) ret++;
	while(ret*ret>v) ret--; return ret;
}
set<pair<int,int>>high[doc];
set<pair<int,int>,greater<pair<int,int>>>low[doc];
int f[moc],v[moc],id; vector<int>son[moc];
inline int find(int p){return f[p]!=p?f[p]=find(f[p]):p;}
int d[moc],fa[moc][20];
inline void dfs(int p,int f){
	fa[p][0]=f; d[p]=d[f]+1;
	for(int i=1;i<=19;++i)
		fa[p][i]=fa[fa[p][i-1]][i-1];
	for(int sp:son[p]) dfs(sp,p);
}
inline int lca(int l,int r){
	if(d[l]<d[r]) swap(l,r);
	for(int i=19;i>=0;i--)
		if(d[fa[l][i]]>=d[r]) l=fa[l][i];
	if(l==r) return l;
	for(int i=19;i>=0;i--)
		if(fa[l][i]!=fa[r][i]) l=fa[l][i],r=fa[r][i];
	return fa[l][0];
}
signed main(){
	if(online)
		freopen("drone.in","r",stdin),
		freopen("drone.out","w",stdout);
	ios::sync_with_stdio(0);
	for(t=io.read();t;t--){
		n=io.read();
		for(int i=1;i<=n;++i) x[i]=io.read(),y[i]=io.read();
		for(int i=1;i<=n;++i) cab[i].clear(),oca[i].clear();
		for(int i=1;i<=n;++i) h[i]=du[i]=0;
		for(int i=1;i<=n*2;++i) son[i].clear();
		for(int i=1;i<=n;++i) low[i].clear();
		for(int i=1;i<=n;++i) high[i].clear();
		m=io.read();
		for(int i=1,l,r,v;i<=m;++i)
			l=io.read(),r=io.read(),
			v=io.read(),tcut[i]={-1,i},
			cab[l].push_back({r,i,v}),
			cab[r].push_back({l,i,v}),
			::l[i]=l,::r[i]=r,
			du[l]++,du[r]++,csd[i]=0;
		for(int i=1;i<=n;++i)
			if(du[i]<=5) q.emplace(i);
		while(q.size()){
			int p=q.front(); q.pop();
			for(const auto&v:cab[p]) if(!csd[v.id]){
				int var=calr(p,v.p,v.v); csd[v.id]=p;
				oca[p].push_back({v.p,v.id,var});
				high[v.p].emplace(var,v.id);
				low[v.p].emplace(-var,v.id);
				// cerr<<p<<" "<<v.p<<" "<<var<<endl;
				// cerr<<v.id<<" "<<p<<endl;
				if(--du[v.p]==5) q.emplace(v.p);
			}
		}
		k=io.read();
		for(int i=1,p,cv;i<=k;++i){
			p=io.read(); cv=io.read(); h[p]+=cv;
			for(const auto&v:oca[p])
				if(tcut[v.id].first==-1){
					high[v.p].erase(make_pair(h[p]-cv+v.v,v.id));
					low[v.p].erase(make_pair(h[p]-cv-v.v,v.id));
					if(abs(h[p]-h[v.p])>v.v) tcut[v.id].first=i;
					else
						high[v.p].emplace(h[p]+v.v,v.id),
						low[v.p].emplace(h[p]-v.v,v.id);
				}
			// cerr<<p<<":\n";
			// for(const auto&v:high[p]) cerr<<v.first<<" "<<v.second<<"   "; cerr<<endl;
			// for(const auto&v:low[p]) cerr<<v.first<<" "<<v.second<<"   "; cerr<<endl;
			while(high[p].size()&&high[p].begin()->first<h[p]){
				pair<int,int> v=*high[p].begin(); high[p].erase(high[p].begin());
				if(tcut[v.second].first==-1) tcut[v.second].first=i;
			}
			while(low[p].size()&&low[p].begin()->first>h[p]){
				pair<int,int> v=*low[p].begin(); low[p].erase(low[p].begin());
				if(tcut[v.second].first==-1) tcut[v.second].first=i;
			}
		}
		// for(int i=1;i<=m;++i)
		// 	cerr<<tcut[i].first<<" "; cerr<<endl;
		iota(f,f+n*2+1,0); id=n;
		for(int i=1;i<=n;++i) v[i]=-1;
		for(int i=1;i<=n;++i)
			for(const auto&v:oca[i])
				if(tcut[v.id].first==-1){
					int l=find(i),r=find(v.p);
					if(l!=r){
						f[l]=f[r]=++id;
						son[id].emplace_back(l);
						son[id].emplace_back(r);
						::v[id]=-1;
					}
				}
		sort(tcut+1,tcut+m+1,greater<pair<int,int>>());
		for(int i=1;i<=m;++i){
			int x=find(l[tcut[i].second]);
			int y=find(r[tcut[i].second]);
			if(x!=y){
				f[x]=f[y]=++id;
				son[id].emplace_back(x);
				son[id].emplace_back(y);
				v[id]=tcut[i].first;
			}
		}
		dfs(id,0); k=io.read();
		for(int i=1,l,r;i<=k;++i)
			l=io.read(),r=io.read(),io.write(v[lca(l,r)],'\n');
	}
}
