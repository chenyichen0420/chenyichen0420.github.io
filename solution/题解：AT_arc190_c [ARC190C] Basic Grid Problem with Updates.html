思路

不会数数，只会暴力。

注意到数据范围只有 $2\times10^5$，于是理所当然的考虑 $O(hw\min(h,w))=O(V\sqrt V)$，思考了一下发现秒了。

不妨钦定 $h\le w$，我们考虑怎么做到单次 $O(h)$。

我们发现，显然我只能枚举这一列，这启示我们是不是可以将从左上角到达这前一列的结果和从右下角到达后一列的结果处理出来，查询的时候直接合并即可？

然后你就发现做完了。上下移动的时候完全不需要更新。如果向左移动就把右边那一列的结果处理出来，如果向右移动就把左边那一列的结果处理出来。

这样修改和查询（虽然是捆绑的）的时间复杂度都是 $O(h)$ 的，足以通过此题。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
constexpr int mod=998244353;
int n,m,k,x,y; char c;
vector<int>a[200005];
vector<int>aur[200005],adl[200005];
signed main(){
	ios::sync_with_stdio(0); cin>>n>>m;
	for(int i=0;i<=n+1;++i) a[i].resize(m+2,0);
	for(int i=0;i<=n+1;++i) aur[i].resize(m+2,0);
	for(int i=0;i<=n+1;++i) adl[i].resize(m+2,0);
	for(int i=1;i<=n;++i)
		for(int j=1;j<=m;++j)
			cin>>a[i][j];
	cin>>k>>x>>y;
	if(n<m){
		for(int i=1;i<=n;++i)
			for(int j=1;j!=y;++j)
				if(i==1&&j==1) adl[i][j]=a[1][1];
				else adl[i][j]=(adl[i-1][j]+adl[i][j-1])*a[i][j]%mod;
		for(int i=n;i>=1;i--)
			for(int j=m;j!=y;--j)
				if(i==n&&j==m) aur[i][j]=a[n][m];
				else aur[i][j]=(aur[i+1][j]+aur[i][j+1])*a[i][j]%mod;
		for(int i=1,v;i<=k;++i){
			cin>>c>>v;
			if(c=='U') x--,a[x][y]=v;
			else if(c=='D') x++,a[x][y]=v;
			else if(c=='L'){
				for(int j=n;j>=1;j--)
					if(j==n&&y==m) aur[j][y]=a[n][m];
					else aur[j][y]=(aur[j+1][y]+aur[j][y+1])*a[j][y]%mod;
				y--; a[x][y]=v;
			}
			else{
				for(int j=1;j<=n;++j)
					if(j==1&&y==1) adl[j][y]=a[1][1];
					else adl[j][y]=(adl[j][y-1]+adl[j-1][y])*a[j][y]%mod;
				y++; a[x][y]=v;
			}
			int sm=0,ans=0;
			if(y==1){
				sm=1;
				for(int j=1;j<=n;++j){
					sm=sm*a[j][1]%mod;
					ans=(ans+sm*aur[j][y+1])%mod;
				}
			}
			else if(y==m){
				sm=1;
				for(int j=n;j>=1;--j){
					sm=sm*a[j][m]%mod;
					ans=(ans+sm*adl[j][y-1])%mod;
				}
			}
			else{
				for(int j=1;j<=n;++j){
					sm=(sm+adl[j][y-1])*a[j][y]%mod;
					ans=(ans+sm*aur[j][y+1])%mod;
				}
			}
			cout<<ans<<endl;
		}
	}
	else{
		for(int i=1;i!=x;++i)
			for(int j=1;j<=m;++j)
				if(i==1&&j==1) adl[i][j]=a[1][1];
				else adl[i][j]=(adl[i-1][j]+adl[i][j-1])*a[i][j]%mod;
		for(int i=n;i!=x;i--)
			for(int j=m;j>=1;--j)
				if(i==n&&j==m) aur[i][j]=a[n][m];
				else aur[i][j]=(aur[i+1][j]+aur[i][j+1])*a[i][j]%mod;
		for(int i=1,v;i<=k;++i){
			cin>>c>>v;
			if(c=='U'){
				for(int j=m;j>=1;--j)
					if(x==n&&j==m) aur[x][j]=a[n][m];
					else aur[x][j]=(aur[x+1][j]+aur[x][j+1])*a[x][j]%mod;
				x--,a[x][y]=v;
			}
			else if(c=='D'){
				for(int j=1;j<=m;++j)
					if(x==1&&j==1) adl[x][j]=a[1][1];
					else adl[x][j]=(adl[x-1][j]+adl[x][j-1])*a[x][j]%mod;
				x++,a[x][y]=v;
			}
			else if(c=='L') y--,a[x][y]=v;
			else y++,a[x][y]=v;
			int sm=0,ans=0;
			if(x==1){
				sm=1;
				for(int j=1;j<=m;++j){
					sm=sm*a[1][j]%mod;
					ans=(ans+sm*aur[x+1][j])%mod;
				}
			}
			else if(x==n){
				sm=1;
				for(int j=m;j>=1;--j){
					sm=sm*a[n][j]%mod;
					ans=(ans+sm*adl[x-1][j])%mod;
				}
			}
			else{
				for(int j=1;j<=m;++j){
					sm=(sm+adl[x-1][j])*a[x][j]%mod;
					ans=(ans+sm*aur[x+1][j])%mod;
				}
			}
			cout<<ans<<endl;
		}
	}
}
