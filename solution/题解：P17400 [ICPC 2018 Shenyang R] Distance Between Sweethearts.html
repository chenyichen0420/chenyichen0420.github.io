思路分析

看见位运算，考虑按位拆贡献进行计算。

于是我们令当前在考虑第 $d$ 位，我们直接枚举 $I_b\in[0,UI_b],I_g\in[0,UI_g]$，统计出 $\max(I_b,I_g)=V,I_b\oplus I_g$ 在第 $d$ 位是 $0/1$ 的方案数 $DPI_{V,0/1}$，同理统计出 $DPA$ 和 $DPG$。

考虑这些 $DP$ 数组的复合。显然直接 $dp1_{v1,x1}\times dp2_{v2,x2}\to newdp_{\max(v1,v2),x1\oplus x2}$ 即可。

将三个 $DP$ 数组复合起来的结果记为 $DP$，那么 $DP_{v,x}$ 对于答案的贡献显然就是 $(\lfloor\frac{v}{2^d}\rfloor\oplus x)\times2^{d}$。这个式子的意义是显然的。

于是我们直接对于每一位都做这个过程即可。时间复杂度 $O(TV^2\log V)$，可以用 $5$ 秒勉强卡过。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int unsigned long long
signed t,xa,ya,za,xb,yb,zb;
int dp[2][2][2005],dc[2][2005],nc[2][2005],ans;
signed main(){
	ios::sync_with_stdio(0); cin>>t;
	for(int tt=1;tt<=t;++tt){
		cin>>xa>>ya>>za>>xb>>yb>>zb;
		for(int d=0;d!=11;++d){
			memset(dp,0,sizeof dp);
			memset(dc,0,sizeof dc);
			for(signed i=0;i<=xa;++i)
				for(signed j=0;j<=xb;++j)
					dc[(i>>d&1)^(j>>d&1)][abs(i-j)]++;
			for(signed i=0;i<=ya;++i)
				for(signed j=0;j<=yb;++j)
					dp[0][(i>>d&1)^(j>>d&1)][abs(i-j)]++;
			for(signed i=0;i<=za;++i)
				for(signed j=0;j<=zb;++j)
					dp[1][(i>>d&1)^(j>>d&1)][abs(i-j)]++;
			for(signed pos=0;pos<=1;++pos){
				memset(nc,0,sizeof nc);
				for(signed ax=0;ax!=2;++ax)
					for(signed av=0;av<=2000;++av)
						for(signed bx=0;bx!=2;++bx)
							for(signed bv=0;bv<=2000;++bv)
								nc[ax^bx][max(av,bv)]+=dc[ax][av]*dp[pos][bx][bv];
				swap(nc,dc);
			}
			for(signed ax=0;ax!=2;++ax)
				for(signed av=0;av<=2000;++av){
					signed at=av; at^=(ax<<d); at=at>>d&1; at<<=d;
					ans+=at*dc[ax][av];
				}
		}
		cout<<"Case #"<<tt<<": "<<ans<<endl; ans=0;
	}
}
