思路分析

这个 $\sum(h+w)\le 1000$ 是干什么的？没看懂。

显然这就是一个缝合题，分成两个独立的问题：

  1. 给定 $n$ 个数字 $a_i$，构造 $b$，满足 $\sum_{i=1}^na_ib_i=0,\operatorname{abs}(b_i)=1$。
  2. 给定 $n$ 个 $x$ 方向的和为 $0$ 的向量和 $n$ 个 $y$ 方向的和为 $0$ 的向量。将他们交错首尾相接，使得任意两个向量不相交。

先看第二问：

横着的和竖着的都被分成了两相等部分，容易想到配对成三个类似直角三角形的形状：

假设路径顺时针构造。

观察蓝色段：如果我让 $x$ 递减，$y$ 递增，则线段斜率递减，于是形成一个上凸包状物，此时所有的点都在斜线及以上。

观察绿色段：如果我让 $x$ 递减，$y$ 递减，则处在那个斜线对应的矩形范围内。

观察红色段，如果我让 $x$ 递减，$y$ 递减，则类似的处在其斜线及以下。

于是这就是一个合法的方案。

如果 $x$ 的上升个数超过了 $y$ 的上升个数，那么交换这两维，类似可以构造。

再看第一问，注意到这就是一道蓝题加上构造方案。直接使用 $O(nV)$ 的背包加上路径记录就行了。不会的可以阅读我的这篇题解。

省流：提前假定一些数字要选，选出来的数字和目标差距为 $\pm V$ 范围内。向后扫描看哪些数字想要，前面哪些数字可以抛弃。由于存在操作顺序满足任意时刻均在 $\pm V$ 范围内，于是时间复杂度就是 $O(nV)$ 的。

于是此题可解，时间复杂度 $O(TnV)$，上不需要保证和在 $10^3$ 内也能过。参考代码如下：

CPP 
#include <bits/stdc++.h>
using namespace std;
int t,n,m,l[1005],r[1005],lp[1005],rp[1005];
pair<int,int>tp[1005]; int dp[1005][2005],cp[1005][2005];
inline void solve(int*v,int*ans,int n){
	for(int i=1;i<=n;++i) tp[i]={v[i],i};
	sort(tp+1,tp+n+1); int sum=0,cur=0,ps=0;
	for(int i=1;i<=n;++i) sum+=tp[i].first;
	if(sum&1) return void(ans[0]=0);
	for(int i=ps=1;i<=n;++i,ps++)
		if((cur+tp[i].first)*2>sum) break;
		else cur+=tp[i].first;
	memset(dp,0,sizeof dp);
	memset(cp,0,sizeof cp);
	sum>>=1; dp[ps-1][cur-sum+1002]=ps;
	for(int i=ps;i<=n;++i){
		memcpy(dp[i],dp[i-1],sizeof dp[0]);
		for(int j=2004;j>tp[i].first;--j)
			if(dp[i-1][j-tp[i].first]>dp[i][j])
				dp[i][j]=dp[i-1][j-tp[i].first],
				cp[i][j]=-i;
		for(int j=2004;j>=1;--j)
			for(int k=dp[i-1][j];k!=dp[i][j];++k)
				if(j>tp[k].first)
					if(dp[i][j-tp[k].first]<k)
						dp[i][j-tp[k].first]=k,
						cp[i][j-tp[k].first]=k;
	}
	if(!dp[n][1002]) return void(ans[0]=0);
	for(int i=1;i!=ps;++i) ans[tp[i].second]=1;
	for(int i=ps;i<=n;++i) ans[tp[i].second]=-1;
	int pc=n,cv=1002;
	while(pc>=ps)
		if(!cp[pc][cv]) pc--;
		else if(cp[pc][cv]>0)
			ans[tp[cp[pc][cv]].second]=-1,
			cv+=tp[cp[pc][cv]].first;
		else
			ans[tp[-cp[pc][cv]].second]=1,
			cv-=tp[-cp[pc][cv]].first, pc--;
	for(int i=1;i<=n;++i) ans[i]*=v[i];
	sum=0;
	for(int i=1;i<=n;++i) sum+=ans[i];
	if(sum!=0) abort();
}
inline bool cmp(int l,int r){
	if(l<0&&r<0) return l>r;
	if(l>0&&r>0) return l<r;
	return l>r;
}
signed main(){
	ios::sync_with_stdio(0);
	for(cin>>t;t;t--){
		cin>>n; lp[0]=1;
		for(int i=1;i<=n;++i) cin>>l[i];
		cin>>m; rp[0]=1;
		for(int i=1;i<=m;++i) cin>>r[i];
		if(n!=m){
			cout<<"No\n"; continue;
		}
		solve(l,lp,n); solve(r,rp,m);
		if(!lp[0]||!rp[0]){
			cout<<"No\n"; continue;
		}
		cout<<"Yes\n"; int clp=0,crp=0,cx=0,cy=0;
		for(int i=1;i<=n;++i) clp+=(lp[i]>0);
		for(int i=1;i<=n;++i) crp+=(rp[i]>0);
		if(clp<=crp){
			sort(lp+1,lp+n+1,greater<int>()),
			sort(rp+1,rp+n+1,cmp);
			for(int i=1;i<=n;++i){
				cx+=lp[i]; cout<<cx<<" "<<cy<<endl;
				cy+=rp[i]; cout<<cx<<" "<<cy<<endl;
			}
		}
		else{
			sort(rp+1,rp+n+1,greater<int>()),
			sort(lp+1,lp+n+1,cmp);
			for(int i=1;i<=n;++i){
				cy+=rp[i]; cout<<cx<<" "<<cy<<endl;
				cx+=lp[i]; cout<<cx<<" "<<cy<<endl;
			}
		}
	}
}
