思路分析

其实看起来非常的善良，但是我一度认为正解时间复杂度不正确。

根据题意，不难设计出一个 DP 状态：$dp_{p,t}$ 表示在 $t$ 时间到达 $p$ 的情况下，在 $p$ 为根的子树内的最小代价。

令 $cnt_p$ 为 $p$ 节点的儿子节点数量，$sp_{p,i}$ 为 $p$ 节点的第 $i$ 个儿子，$[n]$ 表示长度为 $n$ 的排列构成的集合，于是转移是简单的：$\displaystyle dp_{p,t}=c_p(d_p-t)^2+(\min_{perm\in[cnt_p]}\sum_{i=1}^{cnt_p}dp_{sp_{p,perm_i},t+i})$。

后面那个东西，也不难想到将其理解为子结点 $sp$ 匹配到时间 $t$ 会产生 $dp_{sp,t}$ 的代价，于是最小化匹配的代价。

显然这个二分图最小权匹配可以使用 KM 算法解决。但是整体的时间复杂度？

对于单个点，需要计算的 $t$ 的数量为 $O(n)$，每一次计算的时间复杂度为 $cnt^3$，于是总时间复杂度为 $O(\sum n\times cnt^3)\le O(n\times(\sum cnt^3))=O(n^4)$。

显然常数还非常小，于是可以通过此题。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
vector<int>son[205];
int n,c[205],d[205],dp[205][205],sz[205];
struct mcbm{
	int v[205][205],s;
	int hl[205],hr[205];
	bool vl[205],vr[205];
	int pl[205],pr[205];
	int del[205],lp[205];
	inline void flip(int p){
		while(p) pr[p]=lp[p],swap(p,pl[lp[p]]);
	}
	inline void bfs(int p){
		memset(del,0x0f,sizeof del);
		memset(lp,0,sizeof lp);
		memset(vl,0,sizeof vl);
		memset(vr,0,sizeof vr);
		queue<int>q({p});
		while(1){
			while(q.size()){
				p=q.front(); q.pop(); vl[p]=1;
				for(int i=1;i<=s;++i)
					if(!vr[i]&&v[p][i]-hl[p]-hr[i]<del[i]){
						del[i]=v[p][i]-hl[p]-hr[i]; lp[i]=p;
						if(!del[i])
							if(vr[i]=1,!pr[i]) return flip(i);
							else q.emplace(pr[i]);
					}
			}
			int mind=1e9;
			for(int i=1;i<=s;++i)
				if(!vr[i]) mind=min(mind,del[i]);
			if(mind>1e8) return;
			for(int i=1;i<=s;++i)
				if(vl[i]) hl[i]+=mind;
			for(int i=1;i<=s;++i)
				if(vr[i]) hr[i]-=mind;
				else del[i]-=mind;
			for(int i=1;i<=s;++i)
				if(!vr[i]&&!del[i])
					if(vr[i]=1,!pr[i]) return flip(i);
					else q.emplace(pr[i]);
		}
	}
	inline int solve(){
		memset(pl,0,sizeof pl);
		memset(pr,0,sizeof pr);
		memset(hl,0x3f,sizeof hl);
		memset(hr,0,sizeof hr);
		for(int i=1;i<=s;++i)
			for(int j=1;j<=s;++j)
				hl[i]=min(hl[i],v[i][j]);
		for(int i=1;i<=s;++i) bfs(i);
		int ret=0;
		for(int i=1;i<=s;++i) ret+=hl[i]+hr[i];
		return ret;
	}
}km;
inline void dfs(int p,int f,int v){
	int sc=0;
	for(int sp:son[p]) if(sp!=f)
		dfs(sp,p,v+1),sz[p]+=sz[sp],sc++;
	if(!sc){
		for(int t=v;t<=n;++t)
			dp[p][t]=c[p]*(d[p]-t)*(d[p]-t);
		sz[p]=1; return;
	}
	for(int t=v;t<=n-sz[p];++t){
		km.s=sc; int pc=0;
		for(int sp:son[p]) if(sp!=f){
			pc++;
			for(int j=t+1;j<=t+sc;++j)
				km.v[pc][j-t]=dp[sp][j];
		}
		dp[p][t]=km.solve();
	}
	for(int t=0;t<=n-sz[p];++t)
		dp[p][t]+=c[p]*(d[p]-t)*(d[p]-t);
	sz[p]++;
}
signed main(){
	ios::sync_with_stdio(0); cin>>n;
	/*
	设计一个 dp_{p,v} 表示在 v 时刻到达 p，其子树内的最小代价
	于是问题变成了找出子结点代价最小的排列
	*/
	for(int i=1,p;i<=n;++i)
		cin>>c[i]>>d[i]>>p,
		son[p].emplace_back(i),
		son[i].emplace_back(p);
	dfs(0,0,0); cout<<dp[0][0]<<endl;
}

严格限制求解范围之后，常数具体是多小呢？显然我们可以构造出来的最坏情况就是根节点下面挂 $\frac{n}{2}$ 个节点，其中一个节点下面再挂 $\frac{n}{2}$ 个节点。此时的时间复杂度为 $(\frac{n}{2})^4$，显然足以通过此题。
