思路分析

是猫树分治喵。

考虑这里的背包是一个什么样的信息：

支持 $O(V)$ 插入一个物品。不支持删除。

那么，我们会想要使用一些不支持删除的，并且看起来可能需要处理大量信息的，但是查询非常快的数据结构。

考虑到如果只求单点的值，我们可以 $O(V)$ 合并两个背包，因此不难想到猫树分治。

具体来说，我们考虑将询问 $l\neq r$ 标记到猫树上，这样做是 $O(1)$ 的。

所有询问都标记完了之后，我们扫描一遍整棵树。如果区间 $[l,r]$ 的节点上有询问，那么我们处理出 $[l,mid]$ 的后缀背包和 $[mid+1,r]$ 的前缀背包。

查询的时候，直接取出来 $l$ 对应的后缀背包和 $r$ 对应的前缀背包即可。

因此总的时间复杂度是 $O(nV\log n+mV)$，足以通过此题。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,m,s[32768],v[32768],ans[200005],dp[32768][505];
struct cat_tree{
	struct que{ int l,r,v,id; };
	vector<que>q[16][32768];
	inline void ins(int l,int r,int v,int id){
		int d=__lg(l^r)+1,cd=l>>d;
		q[d][cd].push_back({l,r,v,id});
	}
	inline void solve(int l,int r,int id,int d){
		if(l==r||l>n) return;
		int mid=l+r>>1;
		solve(l,mid,id<<1,d-1);
		solve(mid+1,r,id<<1|1,d-1);
		if(!q[d][id].size()) return;
		memset(dp[mid],0xcf,sizeof dp[0]);
		memset(dp[mid+1],0xcf,sizeof dp[0]);
		dp[mid][0]=0; dp[mid][s[mid]]=v[mid];
		dp[mid+1][0]=0; dp[mid+1][s[mid+1]]=v[mid+1];
		for(int i=mid-1;i>=l;i--){
			memcpy(dp[i],dp[i+1],sizeof dp[0]);
			for(int j=500;j>=s[i];--j)
				dp[i][j]=max(dp[i][j],dp[i][j-s[i]]+v[i]);
		}
		for(int i=mid+2;i<=r;++i){
			memcpy(dp[i],dp[i-1],sizeof dp[0]);
			for(int j=500;j>=s[i];--j)
				dp[i][j]=max(dp[i][j],dp[i][j-s[i]]+v[i]);
		}
		for(const auto&v:q[d][id]){
			for(int j=0,mx=0;j<=v.v;++j)
				mx=max(mx,dp[v.l][j]),
				ans[v.id]=max(ans[v.id],mx+dp[v.r][v.v-j]);
		}
	}
}cat;
signed main(){
	ios::sync_with_stdio(0); cin>>n;
	for(int i=0;i!=n;++i) cin>>s[i]>>v[i];
	cin>>m;
	for(int i=1,l,r,v;i<=m;++i)
		if(cin>>l>>r>>v,l--,r--,l==r)
			ans[i]=(v>=s[l]?::v[l]:0);
		else cat.ins(l,r,v,i);
	cat.solve(0,32767,0,15);
	for(int i=1;i<=m;++i) cout<<ans[i]<<endl;
}
