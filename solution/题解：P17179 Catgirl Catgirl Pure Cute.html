思路分析

形式化一下：有一个 $n\times m$ 的基本上全部都是 $1$ 的矩阵，但是有 $k$ 个位置是 $0$，每一次可以将一个 $2\times2$ 的子矩阵的状态翻转了。问你最后最少有多少个 $0$（减一下就是最多多少个 $1$ 了）。

操作是翻转一个 $2\times2$ 的子矩阵，这个操作本身看起来非常的没有前途，我们需要思考我们能够拿他干什么。

一个非常显然的想法是把一个 $2\times2$ 的有超过 $2$ 个“人格猫娘”的子矩阵操作一下，减少数量，但是只能干这件事吗？显然不是。

我们可以举出来反例：一个 $3\times3$ 的 $01$ 矩阵，并且有 $(1,1),(1,3),(3,1)$ 是 $0$。显然我可以对每一个位置都进行一次操作，最后只有 $(3,3)$ 是 $0$。

一个重要的观察是：我可以对 $x\in[x_1,x_2-1],y\in[y_1,y_2-1]$ 的每一个位置进行一次操作，这样做的效果是将 $(x_1,y_1),(x_1,y_2),(x_2,y_1),(x_2,y_2)$ 给翻转了。这是显然的。

那么于是我们考虑怎么样构造最小值？

另一个重要的突破口是找到不变量。上面的操作是选择某两行的某两列翻转掉，则每一行和每一列的 $0$ 的个数的奇偶性不发生变化。

于是有一个显然的下界是有奇数个 $0$ 的行的个数与有奇数个 $0$ 的列的个数的最大值。我们考虑能不能做到这个下界？

一个简单的思路是逐步合法，我们考虑能不能通过一点操作让大部分的行列都满足要求，最后再通过一点操作让剩下的行列也满足要求？

一个非常合理的想法就是将多余的 $0$ 先给赶到边上。具体来说，我们假设有一行有若干个 $0$，我们从中取出两个 $0$，然后把他们赶到第一行。显然这时可以通过矩形的四个角的翻转操作直接完成的。

我们对于每一行都这么做，然后再拿每一列这么做，考虑现在剩下的矩形中有什么性质：

如果某个位置上是 $0$，并且这个位置不在第一行、第一列，那么同行列不会再有别的 $0$，否则会被移动到边上。

说白了，只有原先有奇数个元素的行和列还会剩下元素，别的地方都不会。例外是第一行和第一列。

同时也需要意识到：第一行和第一列也并非完全不满足要求，只是第一行上的元素和第一列上的元素太多了。

我们考虑这些元素应该怎么样移动走，显然我们不能再靠同行列的元素的移动了，于是只能考虑在第一行选择一个元素，在第一列选择一个元素，然后把这部分作为右上角和左下角形成矩形翻转一下。显然前面不能选择 $(1,1)$ 这个点。

当我们终于没有办法进行操作的时候，不难注意到抛开 $(1,1)$ 不看，一定满足第一行没有元素了，或者第一列没有元素了。

我们钦定目前的第一行是没有元素的，那么显然只有第一列中可能会有元素。并且这一列有元素的位置上同一行不再出现别的 $0$。剩下的元素都在中间了。

也就是说，每一个有奇数元素的行都只保留了恰好一个 $0$，而有 $0$ 的列不多于行，有奇数个 $0$ 的行不多于有 $0$ 的行，因此，我们达到了下界。

最后一步，如何快速的翻转一个矩阵内被操作的状态？我们只需要在最后输出一个方案即可，于是直接二维前缀和一下就完成了。

#include <bits/stdc++.h>
using namespace std;
struct IO {
#define mxsz (1 << 21)
	char buf[mxsz], * p1, * p2;
	char pbuf[mxsz], * pp; 
	IO() : p1(buf), p2(buf), pp(pbuf) {}
	~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
	inline char gc() {
		if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, mxsz, stdin);
		return p1 == p2 ? ' ' : *p1++;
	}
	inline int read() {
		int r = 0; char c = gc(); while (c < '0' || c>'9') c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return r;
	}
} io;
int n,m,l,x[1000005],y[1000005],c1,c2,qc;
vector<int>app[1005],cc,cr;
bool cv[1005][1005],col[1005][1005];
int main() {
	ios::sync_with_stdio(0);
	n=io.read(); m=io.read(); l=io.read();
	for (int i=1;i<=l;i++){
		x[i]=io.read(),y[i]=io.read();
		if(n<=1000&&m<=1000)
			app[x[i]].emplace_back(y[i]);
	}
	sort(x+1,x+l+1); sort(y+1,y+l+1);
	qc=0;
	for(int i=1;i<=l;++i)
		if(x[i]!=x[i-1]) c1+=qc&1,qc=1;
		else qc++;
	c1+=qc&1; qc=0;
	for(int i=1;i<=l;++i)
		if(y[i]!=y[i-1]) c2+=qc&1,qc=1;
		else qc++;
	c2+=qc&1;
	cout << n*1ll*m-max(c1, c2) << "\n";
	if(n<=1000&&m<=1000){
		for(int i=2;i<=n;++i){
			while(app[i].size()>1){
				int p1=app[i].back(); app[i].pop_back();
				int p2=app[i].back(); app[i].pop_back();
				if(p1>p2) swap(p1,p2);
				cv[1][p1]=!cv[1][p1];
				cv[1][p2]=!cv[1][p2];
				cv[i][p1]=!cv[i][p1];
				cv[i][p2]=!cv[i][p2];
				app[1].emplace_back(p1);
				app[1].emplace_back(p2);
			}
		}
		for(int i=1;i<=n;++i)
			for(int j:app[i]) col[i][j]=!col[i][j];
		for(int i=1;i<=n;++i) app[i].clear();
		for(int i=1;i<=n;++i)
			for(int j=1;j<=m;++j)
				if(col[i][j]) app[j].emplace_back(i);
		memset(col,0,sizeof col);
		for(int i=2;i<=m;++i){
			while(app[i].size()>1){
				int p1=app[i].back(); app[i].pop_back();
				int p2=app[i].back(); app[i].pop_back();
				if(p1>p2) swap(p1,p2);
				cv[p1][1]=!cv[p1][1];
				cv[p2][1]=!cv[p2][1];
				cv[p1][i]=!cv[p1][i];
				cv[p2][i]=!cv[p2][i];
				app[1].emplace_back(p1);
				app[1].emplace_back(p2);
			}
		}
		for(int i=1;i<=n;++i)
			for(int j:app[i]) col[j][i]=!col[j][i];
		for(int i=2;i<=n;++i)
			if(col[i][1]) cc.emplace_back(i);
		for(int i=2;i<=m;++i)
			if(col[1][i]) cr.emplace_back(i);
		for(int i=0;i!=cc.size()&&i!=cr.size();++i)
			cv[1][1]=!cv[1][1],
			cv[1][cr[i]]=!cv[1][cr[i]],
			cv[cc[i]][1]=!cv[cc[i]][1],
			cv[cc[i]][cr[i]]=!cv[cc[i]][cr[i]];
		for(int i=1;i!=n;++i)
			for(int j=1;j!=m;++j)
				cv[i][j]=cv[i][j]^cv[i][j-1]^cv[i-1][j]^cv[i-1][j-1];
		for(int i=1;i!=n;++i,cout<<endl)
			for(int j=1;j!=m;++j)
				cout<<cv[i][j];
	}
}
