思路分析

这题有紫？先去做弱化版吧。

我们考虑这样一种策略：

我使用 vector<pair<int,int>> 记录一个当前的未遍历过的字典序最小的那种路径的所有结束位置集合以及他们分别的哈希值，并从这个集合出发找下一个。

显然，最有可能能够向后接着走，且字典序最小的位置就是下标最靠前的那个点。显然其他位置的转移位置都是那个点的严格子集，不会比他更优秀。

我们按照转移目标的数值从小到大枚举。权值相同的后继都要放到同一批中进行处理。处理按照深度优先序进行。

这样做的正确性是显然的：本来字典序就是从前向后比较的，我只要有一位比你更小我就一直都比你小了。现在考虑一下时空的问题。

注意到一件事情：如果我添加了一个后继状态，这个后继状态一定会随即被加入到答案当中。

因此，我们不停的加，直到足够输出要求即可。由于每一次继续向下搜索都一定是有值的，所以总的复杂度就是 $O(nV)$ 的，其中 $V$ 为不同的权值个数，此题 $V=n$。所以还不能通过。

考虑到底是哪一步太劣了？就是我找哪些值的转移是存在的，哪些值的转移是不存在的。我们要加速这个过程。

显然，如果我找到的值一定有至少一个转移的话，那么时间复杂度就是 $O(kT)$ 的，其中 $T$ 为找值的时间复杂度。

注意我们前面提到的：最有可能能够向后接着走，且字典序最小的位置就是下标最靠前的那个点，显然其他位置的转移位置都是那个点的严格子集。

也就是说我们只要按照数值大小从小到大枚举了下标最靠前的那个点的所有转移，时间复杂度就是正确的。

而这是简单的，只需要使用一个主席树就能做到。此时的 $T=O(\log n)$，于是时间复杂度为 $O(k\log n)$，足以通过此题。参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
constexpr bool online=0;
struct IO {
#define mxsz (1 << 21)
	char buf[mxsz], * p1, * p2;
	char pbuf[mxsz], * pp; bool acc[128];
	IO() : p1(buf), p2(buf), pp(pbuf) {}
	~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
	inline char gc() {
		if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, mxsz, stdin);
		return p1 == p2 ? ' ' : *p1++;
	}
	inline void setacc(const char* c) { while (*c) acc[*c++] = 1; }
	inline char getc() { char c; while (!acc[c = gc()]); return c; }
#ifndef sipt
	inline int read() {
		int r = 0; char c = gc(); while (c < '0' || c>'9') c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return r;
	}
#else
	inline int read() {
		int r = 0; char c = gc(); bool rev = 0;
		while (c < '0' || c>'9') rev |= (c == '-'), c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return rev ? ~r + 1 : r;
	}
#endif
	inline void push(const char& c) {
		if (pp - pbuf == mxsz) fwrite(pbuf, 1, mxsz, stdout), pp = pbuf;
		*pp++ = c;
	}
	inline void write(int x) {
		static char sta[22]; int top = 0;
		do sta[top++] = x % 10, x /= 10; while (x);
		while (top) push(sta[--top] ^ 48);
	}
	inline void write(int x, char opc) {
#ifdef sopt
		if (x < 0) push('-'), x = ~x + 1;
#endif
		write(x), push(opc);
	}
	inline void puts(const char* c) { while (*c) push(*c++); }
} io;
int n,k,b,m,a[1000005],rt[1000005];
vector<int>ans,app[1000005];
struct seg_tree{
	struct node{ int ls,rs; }re[1000005*30]; int id;
	inline void ins(int cp,int l,int r,int op,int &np){
		re[np=++id]=re[op]; if(l==r) return; int mid=l+r>>1;
		if(cp<=mid) ins(cp,l,mid,re[op].ls,re[np].ls);
		else ins(cp,mid+1,r,re[op].rs,re[np].rs);
	}
}sgt;
struct frame{ int p,l,r; }ft;
inline void dfs(const vector<pair<int,int>>&cur){
	if(!cur.size()) return;
	for(const auto&v:cur) ans.emplace_back(v.second);
	if(ans.size()==k) return;
	auto dfsgt=[&](auto&& self, int p,int l,int r)->void{
		if(!p||ans.size()==k) return;
		if(l==r){
			int lp=upper_bound(app[l].begin(),app[l].end(),cur[0].first)-app[l].begin();
			vector<pair<int,int>>ng;
			for(int i=lp;i!=app[l].size();++i){
				for(const auto&j:cur)
					if(j.first>=app[l][i]) break;
					else{
						ng.emplace_back(app[l][i],(j.second*1ll*b+l)%m);
						if(ng.size()+ans.size()==k) break;
					}
				if(ng.size()+ans.size()==k) break;
			}
			dfs(ng); return;
		}
		int mid=l+r>>1;
		self(self,sgt.re[p].ls,l,mid);
		self(self,sgt.re[p].rs,mid+1,r);
	};
	dfsgt(dfsgt,rt[cur[0].first+1],1,1000000);
}
signed main(){
	if(online)
		freopen("subsequence.in","r",stdin),
		freopen("subsequence.out","w",stdout);
	ios::sync_with_stdio(0);
	n=io.read();k=io.read()+1;b=io.read();m=io.read();
	for(int i=1;i<=n;++i) app[a[i]=io.read()].emplace_back(i);
	for(int i=n;i>=1;i--) sgt.ins(a[i],1,1e6,rt[i+1],rt[i]);
	vector<pair<int,int>>sp; sp.emplace_back(0,0);
	dfs(sp); ans.erase(ans.begin());
	for(int v:ans) io.write(v,'\n');
}
