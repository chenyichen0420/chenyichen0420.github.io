思路分析

值域广，有效值数量与操作数成正比，一眼动态开点线段树优化 DP。

考虑 $dp_{i,j}$ 表示前 $i$ 个操作做完之后数值为 $j$ 的最小代价。考虑转移：

如果遇到赋值，且 $y\neq s$，那么 $dp_{i,p}=dp_{i-1,p}+v(p\neq y)$ 且 $dp_{i,y}=\min dp_{i-1,p}$。如果 $y=s$，那么直接 $dp_{i,p}=dp_{i-1,p}+v$ 即可。

显然这就是全局最小值查询和全局加操作。

如果遇到判断操作，那么我们递归下去做。本质上每一个 if...else 块内都是统一的形式：$x$ 初始值为 $v$，代价为 $a$，求执行完之后 $v$ 是各种值的代价。

于是到达 else 的时候我们就需要将子块的答案合并上来。我们首先将 $v$ 的代价置为 $\inf$，然后将两颗线段树对应位置取最小值合并。

这一步直接用正常的线段树合并就行了。

然后就做完了，时间复杂度是 $O(n\log n)$ 的。参考代码如下：

CPP 
#include <bits/stdc++.h>
using namespace std;
#define int long long
constexpr int inf=1e18;
int n,m,vp[200005][2],jmp[200005]; char op[200005][4]; stack<int>lp; signed rt;
struct seg_tree{
	struct node{
		signed ls,rs; int v,t;
		node():ls(0),rs(0),v(inf),t(0){};
	}re[200005*32]; int id;
	inline void add(int cl,int cr,int cv,int l,int r,signed &p){
		if(!p) p=++id;
		if(cl>=l&&cr<=r) return re[p].v+=cv,void(re[p].t+=cv);
		int mid=l+r>>1;
		if(cl<=mid) add(cl,cr,cv,l,mid,re[p].ls);
		if(cr>mid) add(cl,cr,cv,mid+1,r,re[p].rs);
		re[p].v=min(re[re[p].ls].v,re[re[p].rs].v)+re[p].t;
	}
	inline void set(int cp,int cv,int l,int r,signed&p){
		if(!p) p=++id;
		if(l==r) return void(re[p].v=cv);
		int mid=l+r>>1; cv-=re[p].t;
		if(cp<=mid) set(cp,cv,l,mid,re[p].ls);
		else set(cp,cv,mid+1,r,re[p].rs);
		re[p].v=min(re[re[p].ls].v,re[re[p].rs].v)+re[p].t;
	}
	inline int que(int ql,int qr,int l,int r,signed p){
		if(!p||ql<=l&&qr>=r) return re[p].v;
		int mid=l+r>>1,ret=inf;
		if(ql<=mid) ret=min(ret,que(ql,qr,l,mid,re[p].ls));
		if(qr>mid) ret=min(ret,que(ql,qr,mid+1,r,re[p].rs));
		return ret+re[p].t;
	}
	inline int at(int qp,int l,int r,signed p){
		if(!p||l==r) return re[p].v;
		int mid=l+r>>1,ret=re[p].t;
		if(qp<=mid) ret+=at(qp,l,mid,re[p].ls);
		else ret+=at(qp,mid+1,r,re[p].rs);
		return ret;
	}
	inline void merge(int l,int r,signed &lp,signed op,int cl,int cr){
		if(!op) return;
		if(!lp){
			lp=op; re[op].t+=cr-cl;
			re[op].v+=cr-cl; return;
		}
		if(l==r){
			re[lp].v=min(re[lp].v,re[op].v+cr-cl);
			return;
		}
		int mid=l+r>>1; cl+=re[lp].t; cr+=re[op].t;
		merge(l,mid,re[lp].ls,re[op].ls,cl,cr);
		merge(mid+1,r,re[lp].rs,re[op].rs,cl,cr);
		re[lp].v=min(re[re[lp].ls].v,re[re[lp].rs].v)+re[lp].t;
	}
}sgt;
inline void sim(signed&rt,int adr,int p,int v){
	if(p==m) return; sgt.set(p,v,0,1e9,rt);
	for(int a=adr+1;a!=jmp[adr];++a){
		if(op[a][0]=='i'){
			int pos=vp[a][0],val=sgt.at(pos,0,1e9,rt);
			if(val<inf){
				signed ngs=0; sim(ngs,a,pos,val);
				sgt.set(vp[a][0],inf,0,1e9,rt);
				sgt.merge(0,1e9,rt,ngs,0,0);
			}
			a=jmp[a];
		}
		else{
			if(vp[a][0]==m) sgt.add(0,1e9,vp[a][1],0,1e9,rt);
			else{
				int vm=sgt.que(0,1e9,0,1e9,rt);
				sgt.add(0,1e9,vp[a][1],0,1e9,rt);
				sgt.set(vp[a][0],vm,0,1e9,rt);
			}
		}
	}
}
signed main(){
	ios::sync_with_stdio(0); cin>>n>>m;
	for(int i=1;i<=n;++i)
		if(cin>>op[i],op[i][0]!='e')
			if(cin>>vp[i][0],op[i][0]!='i') cin>>vp[i][1];
			else lp.emplace(i);
		else jmp[lp.top()]=i,lp.pop();
	op[0][0]='i'; op[n+1][0]='e'; jmp[0]=n+1;
	sim(rt,0,0,0); cout<<sgt.re[rt].v<<endl;
}
