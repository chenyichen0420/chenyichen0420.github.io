思路分析

细节决定成败。

首先 $n\le 4$ 的一定要自己手玩一下，你会发现一些边界，这些东西需要特判掉。

考虑剩下的情况，我们正难则反，考虑统计不合法的部分的数量。

容易想到不合法大致有两种情况：

  1. 整张图只有不超过两个颜色。
  2. 删掉某一个点后变成了若干纯色连通块。

我们考虑计算这些东西。

直觉上，对于一个环，其上面只能出现两种颜色，否则必然会失败。然而三元环显然是反例，不过好在这是唯一的反例。因此特判 $n=m=3$ 的情况，输出 $0$。

此外，如果一个三元环向外连接了一条边，那么我们可以让外面的所有边和这个三元环内的对边的颜色一致。此时还能再贡献 $6$ 种情况。显然我们只需要枚举检查有多少个只向外连接一条边的三元环。线性的时空复杂度。

剩下的情况下，每一个环为了保证有解，向外连了边的必须保证内部出现至多两种颜色。

然而如果没有向外连边呢？此时整张图的所有点在同一个点双内。只要 $n\ge 5$，就必须满足出现至多两种颜色。$n=4$ 的又有两个边界情况：$m=6$ 和 $m=5$，下图示意了 $m=6$ 的情况下的一种三色构造：

对于第一种情况，直接点双缩点建立圆方树跑一个 $dp_{i,S}$ 表示以 $i$ 为根的子树内部出现过 $S$ 这个集合内的颜色的方案数，直接暴力转移即可。线性的时空复杂度。

对于第二种情况，考虑直接枚举删除了哪个节点。假设删掉之后剩下了 $c$ 个连通块，要求就是每一种颜色至少出现一次的方案数，显然就是 $3^c-3\times2^c+3$ 种。线性的时空复杂度。

总上此题可解，参考代码如下：

struct IO {
#define mxsz (1 << 20)
	char buf[mxsz], * p1, * p2;
	char pbuf[mxsz], * pp;
	IO() : p1(buf), p2(buf), pp(pbuf) {}
	~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
	inline char gc() {
		if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, mxsz, stdin);
		return p1 == p2 ? ' ' : *p1++;
	}
#ifndef sipt
	inline int read() {
		int r = 0; char c = gc(); while (c < '0' || c>'9') c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return r;
	}
#else
	inline int read() {
		int r = 0; char c = gc(); bool rev = 0;
		while (c < '0' || c>'9') rev |= (c == '-'), c = gc();
		while (c >= '0' && c <= '9') r = r * 10 + (c ^ 48), c = gc();
		return rev ? ~r + 1 : r;
	}
#endif
	inline void push(const char& c) {
		if (pp - pbuf == mxsz) fwrite(pbuf, 1, mxsz, stdout), pp = pbuf;
		*pp++ = c;
	}
	inline void write(int x) {
		static char sta[22]; int top = 0;
		do sta[top++] = x % 10, x /= 10; while (x);
		while (top) push(sta[--top] ^ 48);
	}
	inline void write(int x, char opc) {
#ifdef sopt
		if (x < 0) push('-'), x = ~x + 1;
#endif
		write(x), push(opc);
	}
} io;
int c,t,n,m,pid,dp[600005][8],tmp[8],p2[1000005],p3[1000005],ans;
vector<int>son[300005],ns[600005]; stack<int>s;
int dfn[300005],low[300005],dfc,fa[600005],ec[600005];
inline void tarjan(int p,int f){
	dfn[p]=low[p]=++dfc; s.emplace(p);
	for(int sp:son[p])
		if(!dfn[sp]){
			tarjan(sp,p); low[p]=min(low[p],low[sp]);
			if(low[sp]>=dfn[p]){
				pid++; fa[pid]=p;
				ns[p].emplace_back(pid);
				ns[pid].emplace_back(p);
				while(s.top()!=sp)
					ns[pid].emplace_back(s.top()),
					ns[s.top()].emplace_back(pid),
					fa[s.top()]=pid,s.pop();
				ns[pid].emplace_back(s.top());
				ns[s.top()].emplace_back(pid);
				fa[s.top()]=pid;s.pop();
			}
		}
		else if(sp!=f) low[p]=min(low[p],dfn[sp]);
}
inline int qpow(int a,int b){
	int ret=1;
	for(;b;b>>=1,a=a*a%mod)
		(b&1)&&(ret=ret*a%mod);
	return ret;
}
inline void dfs(int p,int f){
	for(int sp:ns[p]) if(sp!=f) dfs(sp,p);
	if(p>n){
		dp[p][1]=dp[p][2]=dp[p][4]=1;
		dp[p][3]=dp[p][5]=dp[p][6]=p2[ec[p]]-2;
	}
	else{
		dp[p][0]=1;
		ans+=p3[ns[p].size()];
		ans-=p2[ns[p].size()]*3;
		ans+=3;
	}
	for(int sp:ns[p]) if(sp!=f){
		memcpy(tmp,dp[p],sizeof tmp);
		memset(dp[p],0,sizeof dp[0]);
		for(int i=0;i!=8;++i)
			for(int j=0;j!=8;++j)
				dp[p][i|j]=(dp[p][i|j]+tmp[i]*dp[sp][j])%mod;
	}
}
