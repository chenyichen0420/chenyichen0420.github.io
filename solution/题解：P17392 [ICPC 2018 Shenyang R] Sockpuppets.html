思路分析

这种技巧的话，我愿称之为寻找最小限制。

考虑一个非常容易想到的 DP 状态：

$dp_{p,i,j,k}$ 表示在 trie 树上的以 $p$ 节点为根的子树当中，有 $i$ 个未使用的马甲，$j$ 个在到根链上需要一个马甲的作弊者，$k$ 个需要两个马甲的作弊者的方案数。

显然这个东西状态数过大，无法做掉此题。但是我们注意到一个非常强的限制：字符串长度 $\le10$。

也就是说，在到根链上，最多也就只有 $10$ 个潜在的马甲，$10$ 个潜在的作弊者。

于是我们重新设计状态 $dp_{p,i,j,k}$ 表示在以 $p$ 节点为根的子树当中，有 $i$ 个决定要被使用，但暂未使用的马甲，$j$ 个在到根链上需要一个马甲的作弊者，$k$ 个需要两个马甲的作弊者的方案数。

此时，假设 $p$ 的到根链上有 $a$ 个作弊者，$b$ 个马甲，那么显然，只有 $i\le2a,j+2k\le b$ 的状态才是有效的状态。

暴力检验得出，到在 $a+b\le10$ 的情况下，单个点的合法状态数量最多只有 $36$，因此直接暴力做树上 DP 就是正确的。

转移式子是十分好推的，不再赘述，不懂得可以参考代码如下：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
constexpr int mod=1e9+7;
int t,n,m,re[20005][26],rc,rt[20005]; string s;
inline void ins(string&s,int t){
	int cp=0;
	for(char c:s){
		if(!re[cp][c-'a']) re[cp][c-'a']=++rc;
		cp=re[cp][c-'a'];
	}
	rt[cp]=t;
}
vector<vector<vector<int>>>dp[20005],tmp;
inline void mad(int&l,const int r){(l+=r)>=mod&&(l-=mod);}
inline void dfs(int p,int c1,int c2){
	c1+=(rt[p]==1); c2+=(rt[p]==2);
	dp[p].clear(); dp[p].resize(c1<<1|1);
	for(int i=0;i<=(c1<<1);++i){
		dp[p][i].resize(c2+1);
		for(int j=0;j<=c2;++j)
			dp[p][i][j].assign((c2-j>>1)+1,0);
	}
	auto emp=dp[p]; dp[p][0][0][0]=1;
	for(int sp:re[p]) if(sp){
		dfs(sp,c1,c2); auto res=emp; swap(res,dp[p]);
		for(int i=0;i<=(c1<<1);++i)
			for(int j=0;j<=c2;++j)
				for(int k=0;k*2+j<=c2;++k)
					if(res[i][j][k])
						for(int a=0;i+a<=(c1<<1);++a)
							for(int b=0;b+j<=c2;++b)
								for(int c=0;(k+c)*2+b+j<=c2;++c)
									dp[p][i+a][j+b][k+c]=(dp[p][i+a][j+b][k+c]+res[i][j][k]*dp[sp][a][b][c])%mod;
	}
	if(rt[p]==1){
		emp=dp[p]; //我是作弊的，我有 6 类不同的决策
		for(int i=0;i<=(c1<<1);++i)
			for(int j=0;j<=c2;++j)
				for(int k=0;k*2+j<=c2;++k){
					//第一类：啥也不干（那我凭啥是作弊者啊？？），就是原始贡献，已经计入
					//第二类：底下薅一个，于是可用马甲--
					if(i) dp[p][i-1][j][k]=(dp[p][i-1][j][k]+emp[i][j][k]*i)%mod;
					//第三类：底下薅两个，于是可用马甲-=2
					if(i>1) dp[p][i-2][j][k]=(dp[p][i-2][j][k]+emp[i][j][k]*i*(i-1)/2)%mod;
					//第四类：向上天要一份粮，于是单独需求++
					if(j) mad(dp[p][i][j][k],emp[i][j-1][k]);
					//第五类：向上天要两份粮，于是成对需求++
					if(k) mad(dp[p][i][j][k],emp[i][j][k-1]);
					//第六类：上接天宫，下连地府
					if(j&&i) dp[p][i-1][j][k]=(dp[p][i-1][j][k]+emp[i][j-1][k]*i)%mod;
				}
	}
	if(rt[p]==2){
		emp=dp[p]; //我是马甲，我有 4 类不同的决策
		for(int i=0;i<=(c1<<1);++i)
			for(int j=0;j<=c2;++j)
				for(int k=0;k*2+j<=c2;++k){
					//第一类：吃白饭的大肥鱼，就是原始贡献，已经计入
					//第二类：我决定为祖先效力
					if(i) mad(dp[p][i][j][k],emp[i-1][j][k]);
					//第三类：我可以吞掉一个单独马甲需求
					if(j) dp[p][i][j-1][k]=(dp[p][i][j-1][k]+emp[i][j][k]*j)%mod;
					//第四类：我可以砍掉一个双重马甲需求
					if(j!=c2&&k) dp[p][i][j+1][k-1]=(dp[p][i][j+1][k-1]+emp[i][j][k]*k)%mod;
				}
	}
}
signed main(){
	ios::sync_with_stdio(0); cin>>t;
	for(int tt=1;tt<=t;++tt){
		cin>>n>>m; rc=0;
		memset(rt,0,sizeof rt);
		memset(re,0,sizeof re);
		for(int i=1;i<=n;++i) cin>>s,ins(s,1);
		for(int i=1;i<=m;++i) cin>>s,ins(s,2);
		dfs(0,0,0);
		cout<<"Case #"<<tt<<": "<<dp[0][0][0][0]<<endl;
	}
}
