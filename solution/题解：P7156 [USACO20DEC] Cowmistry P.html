思路分析

考虑合法的解的结构：

我们令 $k$ 的最高位是 $2^d$，那么硬性约束合法的解满足三个元素在高于 $d$ 位的地方都是一样的。显然有两种合法的解是在这一位都是 $0$ 或者在这一位都是 $1$。这样两两异或的结果在 $2^d$ 这一位都是 $0$，显然符合要求。

但是还有两种情况：有一个数在 $2^d$ 这一位为 $1$，或者两个数也行。这两种情况是相似的。为了简化，我们这里只讨论有一个数在这一位为 $1$ 的情况。

注意到另外的两个高位为 $0$ 的数字的异或结果在高位是 $0$，因此其实没有限制。

也就是说对于一个高位为 $1$ 的数字 $a$，我们只需要求出出现过的 $a\oplus b$ 的数字个数 $c$，其中 $b\in[2^d,k]$，那么对于答案的贡献就是 $\binom{c}{2}$。

但是，我一来就发现了三个问题：$k$ 的范围是 $10^9$，$a$ 的范围是 $10^9$ 的，检验存在与否还需要时间，你这时间复杂度肯定爆炸！

于是，我们发动俺寻思之力，考虑怎么将这三维一一化解掉。

我们最想解决的就是判断存在性。我如果能够算出来范围，那我是不是就不需要再去检验存不存在，而是去看范围内有多少个数字了呢？

于是问题就变成了我要去求一个范围内的数字异或上一个范围内的数字得到的结果有什么样的计数性质。

我们直接枚举高位为 $0$ 的所有空窗，然后看哪些范围内的高位为 $1$ 的 $a$ 的可行数字包含了这个空窗中的数字，并且扣除相应的个数。即，我们想要快速维护 $\forall a,b,f(a)=\sum_{c=2^d}^k[b\oplus c=a]$。这看起来简直不可做！

继续发动俺寻思之力，考虑一个弱化版的问题：如果 $b$ 是固定的，那么每一个数字最多就被覆盖一次，那么请问 $b\oplus c$ 到底覆盖了哪些数字？

考虑这样一件事情：如果 $c_1$ 和 $c_2$ 只在后 $k$ 位有不同，那么 $b\oplus c$ 也只在后 $k$ 位有不同。如果 $c_0,c_1,\dots,c_{2^k-1}$ 只在后 $k$ 位有不同（实际上就是后 $k$ 位的某种排列了），那么 $b\oplus c$ 就是把 $k$ 位以前的部分固定掉（令这个数字为 $base$）之后，$[base,base+2^k)$ 各出现恰好一次！

于是我们得到一种对于单个数字的快速标记方式：我们对于 $k$，每一次将 $[k+1-\operatorname{lowbit}(k+1),k]$ 处理掉（这些数字就是满足凑齐了最后 $k$ 位不同的部分），然后 $k-\operatorname{lowbit}(k+1)\to k$，重复执行直到 $k=2^d$，注意 $k=2^{d+1}-1$ 的情况需要特殊处理，直接按照上述流程走不到 $k=2^d$。

现在考虑将这个过程推广到一个区间的数字异或一个区间的数字。我们发现仍然可以套用上述流程：利用 $\operatorname{lowbit}$ 将两个区间依次拆分成若干组 $\operatorname{lowbit}$，然后两两组合信息。

现在考虑两组 $\operatorname{lowbit}$ 信息复合。形式化的，我们希望求出来 $[a,a+2^i)$ 中的数字异或 $[b,b+2^j)$ 中的数字的结果对于所有数字的出现次数。其中 $a$ 的低 $i$ 位，$b$ 的低 $j$ 位均为 $0$。

这个东西，不难注意到，令 $i>j$，此时取 $v=a^b$，并将 $v$ 的低 $i$ 位清零得到 $v'$，则被覆盖到的数字是 $[v',v'+2^i)$。覆盖次数是 $2^j$。

我认为这个结论一出来，其实证明就已经充分显然了，所以就不单独证明了。不理解的可以自己打表观察一下二进制（其实就是从第一组数里面翻转一些二进制位来抵消 $b$ 在低位中的出现）。

最后考虑如何统计答案。注意到我们已经解决了有若干空洞的情况下如何正确的计算出每一个数字的实际存在的高位为 $0$ 的数字的数量 $c$。

由于所有的合法数量减都是区间减，所以我们肯定用线段树去维护的这个东西。假设我最后发现高位为 $1$ 的区间 $[l,r]$ 的数字还能对应出来 $c$ 个合法的高位为 $0$ 的数字，那么对于答案的贡献就是 $[l,r]$ 中实际出现了的数的数量 $c1$ 再乘上 $\binom{c}{2}$。

由于一个区间只会拆出来 $O(\log V)$ 个低位区间，而每一个低位区间只会被 $k$ 拆出来 $O(\log V)$ 个低位区间复合区间，而每一个低位区间复合区间都恰好是一个线段数区间，所以时间复杂度就是 $O(n\log^2V\log n+n\log^3V)$。

实质上就是我们把枚举检验存在性变成二分查找区间内数量，把区间异或的枚举按照最低位的完整性拆分成对数级别的小区间，然后恰好发现这样的小区间可以 $O(1)$ 信息复合。

个人认为这个想法还是比较新奇的，虽然放在这道题里面是在不是很优雅。

应该是膨胀成了最复杂，最绕弯子，最劣，最难调的一种解法了。参考代码如下，里面有考场思维链：

CPP 
#include<bits/stdc++.h>
using namespace std;
#define int long long
constexpr int mod=1e9+7,inv6=(mod+1)/6;
int n,k,l[20005],r[20005],s[20005];
vector<int>crp; int lik,wk,val,ans,dk;
inline int c2(int v){ return v*(v-1)/2%mod; }
inline int c3(int v){ return v*(v-1)%mod*(v-2)%mod*inv6%mod; }
inline int inters(int a,int b,int p){
	return max(0ll,min(b,r[p])-max(a,l[p])+1);
}
inline int calc(int a,int b){
    int lp,rp,mid,p1,p2,ret=0;
    lp=1,rp=n+1;
    while(lp!=rp){
        mid=lp+rp>>1;
        if(r[mid]<a)lp=mid+1;
        else rp=mid;
    }
    p1=lp; lp=0,rp=n;
    while(lp!=rp){
        mid=lp+rp+1>>1;
        if(l[mid]<=b)lp=mid;
        else rp=mid-1;
    }
    p2=lp;
    if(p1>p2) return 0;
    if(p1==p2) return inters(a,b,p1);
    ret=s[p2-1]-s[p1];
    ret+=inters(a,b,p1);
    ret+=inters(a,b,p2);
    return ret;
}
struct seg_tree{
	struct node{ signed ls,rs,v; }re[20005<<10]; int cnt;
	inline void clear(){
		for(int i=0;i<=cnt;++i) re[i]={0,0,0};
		cnt=0;
	}
	inline void ins(int cl,int cr,int cv,int l,int r,signed& p){
		if(!p) p=++cnt; int mid=l+r>>1;
		if(cl<=l&&cr>=r) return void(re[p].v+=cv);
		if(cl<=mid) ins(cl,cr,cv,l,mid,re[p].ls);
		if(cr>mid) ins(cl,cr,cv,mid+1,r,re[p].rs);
	}
	inline void dfs(int l,int r,int c,int p){
		if(c+=re[p].v,!p||l==r){
			ans=(ans+calc(l,r)*c2(dk-c))%mod;
			return;
		}
		int mid=l+r>>1;
		dfs(l,mid,c,re[p].ls);
		dfs(mid+1,r,c,re[p].rs);
	}
}sgt;
inline vector<pair<int,int>> ints(int a,int b){
    vector<pair<int,int>> ret;
    int lp=1,rp=n+1;
    while(lp!=rp){
        int mid=(lp+rp)>>1;
        if(r[mid]<a) lp=mid+1;
        else rp=mid;
    }
    int cp=a;
    while(lp<=n&&l[lp]<=b){
        if(cp<l[lp]) ret.emplace_back(cp,l[lp]-1);
        cp=max(cp,r[lp]+1); lp++;
    }
    if(cp<=b) ret.emplace_back(cp,b);
    return ret;
}
int ca,cb; signed crt;
inline void markk(int l,int r){
    int tk=k-(1<<wk)+1;
    l^=(1<<wk); r^=(1<<wk);
    int lenx=r-l+1;
    while(tk){
        int pc=tk&-tk; tk-=pc;
        int mask=max(lenx,pc)-1;
        int ll=l^tk;
        int pol=ll&~mask;
        int por=pol+mask;
        sgt.ins(pol,por,min(lenx,pc),ca,cb,crt);
    }
}
inline void markseg(int l,int r){
	r++;
	while(l){
		int lb=l&-l;
		if(l+lb<=r)
			markk(l,l+lb-1),
			l+=lb;
		else break;
	}
	while(r){
		int lb=r&-r; r-=lb;
		if(r>=l) markk(r,r+lb-1);
		else break;
	}
}
inline void solve(int a,int b){
	int mid=a+b>>1; ans=(ans+c3(calc(a,mid))+c3(calc(mid+1,b)))%mod;
	vector<pair<int,int>>intl=ints(a,mid),intr=ints(mid+1,b);
	ca=mid+1; cb=b; sgt.clear(); crt=0;
	for(const auto&v:intl) markseg(v.first,v.second);
	sgt.dfs(ca,cb,0,crt);
	ca=a; cb=mid; sgt.clear(); crt=0;
	for(const auto&v:intr) markseg(v.first,v.second);
	sgt.dfs(ca,cb,0,crt);
}
signed main(){
	ios::sync_with_stdio(0); cin>>n>>k;
	if(k<=2) return puts("0"),0;
	for(int i=1;i<=n;++i) cin>>l[i]>>r[i];
	/*
	如果选择的三种试剂均只在 __lg(k)-1 或者更低位不同，那么这些组合全部都满足要求
	这种组合的数量可以 O(n poly(log)) 快速计算，不在瓶颈上

	也就是说现在的三种试剂必然在第 __lg(k) 位上出现了不同

	出现了不同则必定是一个高位为 0，两个高位为 1，或者反之
	两个高位相同的数字天然满足要求，因此核心在于低位的那个数字
	如果高位的数字是齐的，那么显然满足要求的数字个数就是 k 的低位部分+1

	能不能给这个东西建一个虚拟的 trie 树出来？
	这个 trie 树直接维护区间信息？做不到
	重点在于不同的数字的修改都会被拆分成至多 O(\log) 个区间
	而且这些区间还没有任何联系，无法快速标记

	这些区间似乎是很有可能重复的，考虑有没有拆分成 log 组区间的希望
	具体的拆分方式是每一次剥离 lowbit，原因：
	假设我们现在正在钦定高位有一个 0
	那么我就需要找出这个 0 的数字有多少个合法的高位为 1 的数字
	于是我们反过来，枚举高位的不合法部分，然后去给相应的低位扣除贡献
	需要被扣除的就是所有异或上低位 <=k 的数值的结果
	
	如果我每一次把 lowbit 相同的部分的结果都给砍掉
	那么这些数字在 lowbit 以上的部分是写死的
	lowbit 以下的部分是被均匀涂抹的
	因此这些区间的标记可以一次性大上

	由于空窗数量是 O(n) 的
	每一个空窗只会被拆分为 O(\log) 组标记区间
	每组再被 k 按照 lowbit 拆分成 O(\log) 个实际标记区间

	因此总的时间复杂度是 O(n\log n\log^2 V) 的！


	思考一下细节怎么处理：
	我们将值域拆分成 2^{__lg(k)+1} 的段进行从处理
	每一段有三种可能的状态：
	部分可用，完全可用，完全不可用
	完全可用的部分可以直接 O(n) 计算总的贡献
	部分可用的可以直接计算出来 O(n) 个区间，然后用上面的方法进行处理
	完全不可用的部分就直接不管就行了

	看起来实现会是这道题的另一个难点
	*/
	lik=1<<(__lg(k)+1); wk=__lg(k);
	for(int i=1;i<=n;++i)
		crp.emplace_back(l[i]>>wk+1),
		crp.emplace_back(r[i]>>wk+1);
	sort(crp.begin(),crp.end()); dk=k-(1<<wk)+1;
	crp.erase(unique(crp.begin(),crp.end()),crp.end());
	val=(c3(1<<wk)*2+(c2(k-(1<<wk)+1)<<wk+1))%mod;
	for(int i=1;i<=n;++i){
		int cnt=(r[i]>>wk+1)-(l[i]>>wk+1)-1;
		if(cnt>0) ans=(ans+cnt*val)%mod;
	}
	for(int i=1;i<=n;++i) s[i]=r[i]-l[i]+1;
	for(int i=1;i<=n;++i) s[i]+=s[i-1];
	for(int i:crp) solve(i<<wk+1,(i+1<<wk+1)-1);
	cout<<ans<<endl;
}
